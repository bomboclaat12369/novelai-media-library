// Shared membership indicators and explicit collection assignment. No polling.
(() => {
  'use strict';
  const root=document.getElementById('nai-media-host')?.shadowRoot;
  if(!root)return;
  const kind=s=>s.kind==='connection'?'Connection':'Set';
  const style=document.createElement('style');
  style.textContent=`
    #naiMemberships{display:flex;gap:5px;flex-wrap:wrap;align-items:center;padding:5px 10px;font:11px system-ui}
    #naiMemberships[hidden],#naiMemberships.hidden{display:none}
    #naiMemberships button{font-size:11px;padding:4px 7px;max-width:100%;overflow-wrap:anywhere}
    .naiGroupBadge,.naiMainBadge{position:absolute;z-index:3;background:rgba(8,10,25,.88);border-radius:4px;padding:2px 4px;font:700 15px system-ui;color:#b8b0ff;line-height:1.1}
    .naiGroupBadge{bottom:5px;right:5px}.naiMainBadge{top:5px;left:39px;color:#a1e1d0}
    .naiSetCount{font-size:9px!important}.naiSetCoverBtn[aria-pressed="true"]{color:#ffdb66}
    .naiGroupAssign{display:flex;flex-direction:column;gap:12px}
  `;
  root.appendChild(style);
  let snapshot=null, index=new Map();
  function memberships(id,lib){
    if(lib!==snapshot){snapshot=lib;index=new Map();for(const s of lib?.sets || [])for(const mid of s.media_ids || []){if(!index.has(mid))index.set(mid,[]);index.get(mid).push(s);}}
    return index.get(id)||[];
  }
  root.naiPaintGroupBadge=(tile,m,lib)=>{
    const groups=memberships(m.id,lib);
    if(!groups.length)return;
    const badge=document.createElement('span');badge.className='naiGroupBadge';badge.textContent='▧';
    badge.title=groups.map(s=>`${kind(s)}: ${s.name}`).join('\n');badge.setAttribute('aria-label',badge.title);tile.appendChild(badge);
  };
  root.naiRenderMemberships=(m,lib)=>{
    let panel=root.getElementById('naiMemberships');
    if(!panel){panel=document.createElement('div');panel.id='naiMemberships';root.getElementById('qualityInfo').after(panel);}
    panel.replaceChildren();panel.hidden=!m;
    if(!m)return;
    for(const s of memberships(m.id,lib)){
      const b=document.createElement('button');b.textContent=`▧ ${kind(s)}: ${s.name}`;b.dataset.groupId=s.id;
      b.title=`Open ${kind(s).toLowerCase()} “${s.name}”`;b.onclick=()=>root.naiOpenGroup(s.id);panel.appendChild(b);
    }
    const connect=document.createElement('button');connect.textContent='Connect…';connect.id='naiConnectMedia';
    connect.onclick=()=>root.naiAssignGroups([m.id],'connection');panel.appendChild(connect);
  };
  function request(path,body){return new Promise((resolve,reject)=>GM_xmlhttpRequest({method:body?'POST':'GET',url:'http://127.0.0.1:8765'+path,
    headers:{Origin:'https://novelai.net','Content-Type':'application/json'},data:body?JSON.stringify(body):null,timeout:30000,
    onload:r=>{let v;try{v=JSON.parse(r.responseText);}catch(_){reject(Error('Invalid companion response'));return;}if(r.status>=200&&r.status<300)resolve(v);else reject(Error(v.error||'Could not save collection'));},
    onerror:()=>reject(Error('Could not connect to the local companion.')),ontimeout:()=>reject(Error('The companion did not respond.'))}));}
  root.naiRequireCollections=async()=>{
    const health=await request('/api/health');
    if(!health.mixed_collections)throw Error('Restart the companion to install Sets & Connections support, then try again.');
  };
  root.naiAssignGroups=async(ids,preferred='set')=>{
    try{
      await root.naiRequireCollections();
      const lib=await request('/api/library');
      const media=ids.map(id=>lib.media.find(m=>m.id===id)).filter(Boolean);
      if(!media.length)throw Error('Select saved media first.');
      const characterId=media[0].character_id;
      root.naiPrepareGroupOwner(lib,characterId);
      if(media.some(m=>m.character_id!==characterId))throw Error('Choose media from one character.');
      const wrap=document.createElement('div');wrap.className='modalWrap';
      wrap.innerHTML='<div class="modal"><div class="modalHead"><div class="modalTitle">Add to Set / Connection</div><button data-close>×</button></div><div class="modalBody naiGroupAssign"><div class="muted">Adding to a Set preserves main Images placement and moves membership from any previous Set. Connections preserve all existing memberships.</div><select id="naiAssignKind" aria-label="Collection type"><option value="set">Set</option><option value="connection">Connection</option></select><select id="naiAssignExisting" aria-label="Existing collection"></select><button id="naiAssignAdd" class="primary">Add to selected</button><button id="naiAssignNew">Create new and choose media…</button><div id="naiAssignStatus" role="status"></div></div></div>';
      const modal=root.getElementById('modalRoot');modal.replaceChildren(wrap);
      const type=wrap.querySelector('#naiAssignKind'),select=wrap.querySelector('#naiAssignExisting'),add=wrap.querySelector('#naiAssignAdd');
      const paint=()=>{select.replaceChildren();for(const s of lib.sets||[])if(s.character_id===characterId&&(s.kind||'set')===type.value){const o=document.createElement('option');o.value=s.id;o.textContent=s.name;select.appendChild(o);}add.disabled=!select.options.length;};
      type.value=preferred;type.onchange=paint;paint();
      const close=()=>wrap.remove();wrap.querySelector('[data-close]').onclick=close;wrap.onclick=e=>{if(e.target===wrap)close();};
      wrap.querySelector('#naiAssignNew').onclick=()=>root.naiOpenGroupManager(null,ids,type.value);
      add.onclick=async()=>{add.disabled=true;try{
        await request('/api/sets/'+encodeURIComponent(select.value),{add_media_ids:ids,preserve_placement:true});
        const updated=await request('/api/library');
        close();root.naiRenderNormalGrid(updated);root.naiSetsApplyLibrary216(updated);root.naiRefreshSelectedMetadata();
      }catch(e){wrap.querySelector('#naiAssignStatus').textContent=e.message;add.disabled=false;}};
    }catch(e){alert(e.message);}
  };
  root.naiRefreshSelectedMetadata?.();
  root.naiRefreshGrid?.();
})();
