
// NovelAI Media Library v2.3.6: right-panel alignment + reliable video thumbnail refresh
(() => {
  'use strict';

  const API = 'http://127.0.0.1:8765';
  const host = document.getElementById('nai-media-host');
  const root = host?.shadowRoot;
  if (!root || document.documentElement.dataset.naiMediaV236 === '1') return;
  document.documentElement.dataset.naiMediaV236 = '1';

  // From the user's 2048px-wide screenshot, NovelAI's story scrollbar sits at ~1470px.
  // 72.1vw places the media panel only a few pixels to its right, matching the small
  // gap between the left media panel and the start of the story text.
  const style = document.createElement('style');
  style.id = 'nai-media-v236-style';
  style.textContent = `
    :host { --nai-right-edge: 72.1vw !important; }
  `;
  root.appendChild(style);

  function gmRequest({ method = 'GET', path, data = null, headers = {}, responseType = 'text', timeout = 300000 }) {
    return new Promise((resolve, reject) => {
      GM_xmlhttpRequest({
        method,
        url: `${API}${path}`,
        data,
        headers: { Origin: 'https://novelai.net', ...headers },
        responseType,
        timeout,
        onload: res => {
          if (res.status >= 200 && res.status < 300) {
            if (responseType === 'blob') return resolve(res.response);
            const text = res.responseText || '';
            if (!text) return resolve(null);
            try { resolve(JSON.parse(text)); } catch (_) { resolve(text); }
            return;
          }
          let message = `Request failed (${res.status})`;
          try { message = JSON.parse(res.responseText || '{}').error || message; } catch (_) {}
          reject(new Error(message));
        },
        onerror: () => reject(new Error('Could not connect to the local NovelAI Media Library program.')),
        ontimeout: () => reject(new Error('The local media request timed out.')),
      });
    });
  }

  function waitFor(video, eventName, timeout = 30000) {
    return new Promise((resolve, reject) => {
      let timer = 0;
      const cleanup = () => {
        clearTimeout(timer);
        video.removeEventListener(eventName, done);
        video.removeEventListener('error', failed);
      };
      const done = () => { cleanup(); resolve(); };
      const failed = () => { cleanup(); reject(new Error('Could not load the video frame for thumbnail capture.')); };
      timer = setTimeout(() => {
        cleanup();
        reject(new Error(`Timed out waiting for video ${eventName}.`));
      }, timeout);
      video.addEventListener(eventName, done, { once: true });
      video.addEventListener('error', failed, { once: true });
    });
  }

  async function captureFrame(mediaId, seconds) {
    const sourceBlob = await gmRequest({ path: `/api/media/${mediaId}/original`, responseType: 'blob' });
    const sourceUrl = URL.createObjectURL(sourceBlob);
    const video = document.createElement('video');
    video.preload = 'auto';
    video.muted = true;
    video.playsInline = true;
    video.src = sourceUrl;

    try {
      if (video.readyState < 1) await waitFor(video, 'loadedmetadata');
      const duration = Number.isFinite(video.duration) ? video.duration : 0;
      const target = duration > 0
        ? Math.min(Math.max(0, seconds), Math.max(0, duration - 0.01))
        : Math.max(0, seconds);

      if (Math.abs(video.currentTime - target) > 0.001) {
        video.currentTime = target;
        await waitFor(video, 'seeked');
      } else if (video.readyState < 2) {
        await waitFor(video, 'loadeddata');
      }

      if (!video.videoWidth || !video.videoHeight) throw new Error('Video frame is not ready yet.');
      const maxSide = 960;
      const scale = Math.min(1, maxSide / Math.max(video.videoWidth, video.videoHeight));
      const canvas = document.createElement('canvas');
      canvas.width = Math.max(1, Math.round(video.videoWidth * scale));
      canvas.height = Math.max(1, Math.round(video.videoHeight * scale));
      const ctx = canvas.getContext('2d');
      if (!ctx) throw new Error('Could not capture this video frame.');
      ctx.drawImage(video, 0, 0, canvas.width, canvas.height);

      return await new Promise((resolve, reject) => {
        canvas.toBlob(
          blob => blob ? resolve(blob) : reject(new Error('Could not create thumbnail image.')),
          'image/jpeg',
          0.9
        );
      });
    } finally {
      video.pause();
      video.removeAttribute('src');
      video.load();
      URL.revokeObjectURL(sourceUrl);
    }
  }

  function activeVideo() {
    return root.querySelector('.stage.slot-active video') || root.querySelector('.stage:not(.hidden) video');
  }

  function mediaIdForVideo(video) {
    if (video) {
      const source = String(video.currentSrc || video.src || '');
      const match = source.match(/\/api\/media\/([0-9a-f]+)\/original(?:$|[?#])/i);
      if (match?.[1]) return match[1];
    }
    return root.querySelector('.tile.active')?.dataset?.id || null;
  }

  // The base UI keeps thumbnail object URLs in an internal cache. Updating the file on
  // disk alone can therefore leave the old frame on-screen until a full page reload.
  // Keep a tiny override cache for thumbnails changed during this page session and apply
  // the exact JPEG blob we just saved, so the new frame is visible immediately.
  const changedThumbUrls = new Map();

  function applyChangedThumb(mediaId) {
    const url = changedThumbUrls.get(mediaId);
    if (!url) return;
    const tile = root.querySelector(`.tile[data-id="${mediaId}"]`);
    if (!tile) return;

    let img = tile.querySelector('img[data-mid]');
    if (!img) {
      img = document.createElement('img');
      img.loading = 'eager';
      img.dataset.mid = mediaId;
      tile.insertBefore(img, tile.firstChild);
    }
    if (img.src !== url) img.src = url;
    img.alt = '';
  }

  function rememberChangedThumb(mediaId, blob) {
    const old = changedThumbUrls.get(mediaId);
    if (old) URL.revokeObjectURL(old);
    const url = URL.createObjectURL(blob);
    changedThumbUrls.set(mediaId, url);
    applyChangedThumb(mediaId);
  }

  function reapplyChangedThumbsSoon() {
    if (!changedThumbUrls.size) return;
    setTimeout(() => {
      for (const mediaId of changedThumbUrls.keys()) applyChangedThumb(mediaId);
    }, 0);
  }

  // v2.3.0's thumbnail handler is on document capture and stops propagation. Window
  // capture runs first, so this safely replaces only that button behavior without a DOM
  // observer or any continuous work.
  window.addEventListener('click', async event => {
    const path = event.composedPath?.() || [];
    const button = path.find(node => node?.id === 'setThumbBtn');
    if (!button || !path.includes(host)) return;

    event.preventDefault();
    event.stopImmediatePropagation();

    const video = activeVideo();
    const mediaId = mediaIdForVideo(video);
    if (!video || !mediaId) {
      alert('Could not determine which video is active. Select the video thumbnail again, then retry.');
      return;
    }

    button.disabled = true;
    button.textContent = 'Saving thumbnail…';
    try {
      const seconds = Number.isFinite(video.currentTime) ? video.currentTime : 0;
      const frameBlob = await captureFrame(mediaId, seconds);
      await gmRequest({
        method: 'POST',
        path: `/api/media/${mediaId}/thumbnail?seconds=${encodeURIComponent(seconds)}`,
        headers: { 'Content-Type': 'image/jpeg' },
        data: frameBlob,
      });

      // Use the exact frame blob just written instead of refetching through the old
      // in-page thumbnail cache.
      rememberChangedThumb(mediaId, frameBlob);
      button.textContent = 'Thumbnail saved';
    } catch (err) {
      alert(err?.message || String(err));
      button.textContent = 'Set thumbnail to current frame';
    } finally {
      setTimeout(() => {
        if (!button.isConnected) return;
        button.disabled = false;
        if (button.textContent === 'Thumbnail saved') button.textContent = 'Set thumbnail to current frame';
      }, 700);
    }
  }, true);

  // If the grid is rebuilt by changing category/character/search, put any thumbnail
  // changed during this session back on the newly-created tile.
  root.addEventListener('click', reapplyChangedThumbsSoon, false);
  root.addEventListener('change', reapplyChangedThumbsSoon, false);
  root.addEventListener('input', reapplyChangedThumbsSoon, false);

  window.addEventListener('beforeunload', () => {
    for (const url of changedThumbUrls.values()) URL.revokeObjectURL(url);
    changedThumbUrls.clear();
  }, { once: true });
})();
