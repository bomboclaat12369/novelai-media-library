
// NovelAI Media Library v2.3.5: thumbnail display, right-panel width, split fullscreen
(() => {
  'use strict';

  const host = document.getElementById('nai-media-host');
  const root = host?.shadowRoot;
  if (!root || document.documentElement.dataset.naiMediaV235 === '1') return;
  document.documentElement.dataset.naiMediaV235 = '1';

  const style = document.createElement('style');
  style.id = 'nai-media-v235-style';
  style.textContent = `
    /* Leave more horizontal room for NovelAI's story text. */
    :host { --nai-right-edge: 74vw !important; }

    /* A video with a real thumbnail already has an <img>. Keep only a small VIDEO badge
       instead of covering that image with the full blue placeholder. */
    .tile > img[data-mid] ~ .videoMark {
      inset: auto auto 5px 5px !important;
      width: auto !important;
      height: auto !important;
      max-width: calc(100% - 10px) !important;
      min-height: 0 !important;
      display: block !important;
      padding: 2px 6px !important;
      border-radius: 999px !important;
      background: rgba(0,0,0,.68) !important;
      pointer-events: none !important;
      overflow: hidden !important;
      text-overflow: ellipsis !important;
      white-space: nowrap !important;
      font: 700 10px system-ui !important;
    }
    .tile > img[data-mid] ~ .videoMark .videoName { display:none !important; }

    .naiSplitFullscreen {
      position: fixed !important;
      inset: 0 !important;
      z-index: 2147483950 !important;
      background: rgba(0,0,0,.96) !important;
      display: grid !important;
      grid-template-columns: repeat(2, minmax(0, 1fr)) !important;
      gap: 8px !important;
      padding: 18px 24px !important;
    }
    .naiSplitFullscreen.single { grid-template-columns: minmax(0,1fr) !important; }
    .naiSplitFullscreenPane {
      min-width: 0 !important;
      min-height: 0 !important;
      display: flex !important;
      align-items: center !important;
      justify-content: center !important;
      overflow: hidden !important;
      background: #05060d !important;
    }
    .naiSplitFullscreenPane img,
    .naiSplitFullscreenPane video {
      width: 100% !important;
      height: 100% !important;
      max-width: 100% !important;
      max-height: 100% !important;
      object-fit: contain !important;
      display: block !important;
    }
    .naiSplitFullscreenClose {
      position: fixed !important;
      top: 12px !important;
      right: 14px !important;
      z-index: 2147483960 !important;
      width: 40px !important;
      height: 40px !important;
      padding: 0 !important;
      display: grid !important;
      place-items: center !important;
      border-radius: 50% !important;
      background: rgba(22,24,47,.88) !important;
      font-size: 24px !important;
    }
  `;
  root.appendChild(style);

  function closeSplitFullscreen() {
    const lightRoot = root.getElementById('lightRoot');
    if (!lightRoot) return;

    lightRoot.querySelectorAll('video[data-nai-split-full]').forEach(fullVideo => {
      const original = fullVideo._naiOriginalVideo;
      if (original?.isConnected) {
        try { original.currentTime = fullVideo.currentTime; } catch (_) {}
        if (!fullVideo.paused) original.play().catch(() => {});
        else original.pause();
      }
      try { fullVideo.pause(); } catch (_) {}
    });

    lightRoot.innerHTML = '';
  }

  function copyForFullscreen(source) {
    if (source.tagName === 'IMG') {
      const img = document.createElement('img');
      img.src = source.currentSrc || source.src;
      img.alt = source.alt || '';
      return img;
    }

    if (source.tagName === 'VIDEO') {
      const video = document.createElement('video');
      const startTime = Number.isFinite(source.currentTime) ? source.currentTime : 0;
      const wasPlaying = !source.paused;

      video.dataset.naiSplitFull = '1';
      video.controls = true;
      video.loop = true;
      video.playsInline = true;
      video.volume = Number.isFinite(source.volume) ? source.volume : 0.08;
      video.muted = source.muted;
      video._naiOriginalVideo = source;
      video.src = source.currentSrc || source.src;

      source.pause();
      video.addEventListener('loadedmetadata', () => {
        try {
          const end = Number.isFinite(video.duration) && video.duration > 0
            ? Math.max(0, video.duration - 0.05)
            : startTime;
          video.currentTime = Math.min(startTime, end);
        } catch (_) {}
        if (wasPlaying) video.play().catch(() => {});
      }, { once: true });
      return video;
    }

    return null;
  }

  function openSplitFullscreen() {
    const stages = root.getElementById('stages');
    if (!stages?.classList.contains('split')) return false;

    const sources = [root.getElementById('stage1'), root.getElementById('stage2')]
      .map(stage => stage?.querySelector('img,video'))
      .filter(Boolean)
      .slice(0, 2);
    if (!sources.length) return false;

    const lightRoot = root.getElementById('lightRoot');
    if (!lightRoot) return false;
    lightRoot.innerHTML = '';

    const box = document.createElement('div');
    box.className = `naiSplitFullscreen${sources.length === 1 ? ' single' : ''}`;

    for (const source of sources) {
      const pane = document.createElement('div');
      pane.className = 'naiSplitFullscreenPane';
      const copy = copyForFullscreen(source);
      if (copy) pane.appendChild(copy);
      box.appendChild(pane);
    }

    const close = document.createElement('button');
    close.type = 'button';
    close.className = 'naiSplitFullscreenClose';
    close.textContent = '×';
    close.title = 'Close';
    close.addEventListener('click', event => {
      event.preventDefault();
      event.stopPropagation();
      closeSplitFullscreen();
    });
    box.appendChild(close);
    box.addEventListener('click', event => {
      if (event.target === box) closeSplitFullscreen();
    });

    lightRoot.appendChild(box);
    return true;
  }

  /* Capture before the v2.3.0 single-item fullscreen handlers. In split mode,
     fullscreen means both visible slots side by side. */
  root.addEventListener('click', event => {
    const stages = root.getElementById('stages');
    if (!stages?.classList.contains('split')) return;

    const target = event.target;
    const fullButton = target?.closest?.('#fullBtn');
    const stageImage = target?.closest?.('.stage img');
    if (!fullButton && !stageImage) return;

    event.preventDefault();
    event.stopImmediatePropagation();
    openSplitFullscreen();
  }, true);

  document.addEventListener('keydown', event => {
    if (event.key === 'Escape' && root.querySelector('.naiSplitFullscreen')) {
      closeSplitFullscreen();
    }
  }, true);
})();
