  const formatBytes = bytes => {
    const value=Number(bytes)||0;

    const units=['KB','MB','GB','TB']; let n=value/1024, i=0;

  function gmRequest({method='GET', path, data=null, headers={}, responseType='text', timeout=120000}) {
    return new Promise((resolve, reject) => GM_xmlhttpRequest({
      method,
      url:`${API}${path}`,
      data,
      headers:{Origin:'https://novelai.net', ...headers},
      responseType,
      timeout,
      onload:res => {
        if (res.status >= 200 && res.status < 300) {
          if (responseType === 'blob') return resolve(res.response);
          const text = res.responseText || '';
          if (!text) return resolve(null);
          try { return resolve(JSON.parse(text)); } catch (_) { return resolve(text); }
        }
        let message = `Request failed (${res.status})`;
        try { message = JSON.parse(res.responseText || '{}').error || message; } catch (_) {}
        reject(new Error(message));
      },
      onerror:()=>reject(new Error('Could not connect to the local NovelAI Media Library program.')),
      ontimeout:()=>reject(new Error('The local media request timed out.')),
    }));
  }

    // Use Tampermonkey's localhost-capable request path directly. Native page fetches from

  async function sha256File(file) {
    if (!file || file.size > 128 * 1024 * 1024) return null;
    try {
      const digest = await crypto.subtle.digest('SHA-256', await file.arrayBuffer());
      return [...new Uint8Array(digest)].map(x => x.toString(16).padStart(2, '0')).join('');
    } catch (_) { return null; }
  }

        });
        if (item) {
          const duplicate = beforeIds.has(item.id);
          if (!duplicate) return {item, duplicate:false, confirmed:true};

          const required = Array.isArray(categories) ? categories : [];
          const beforeCats = Array.isArray(beforeItem?.categories) ? beforeItem.categories : [];

          const noChangeNeeded = required.every(id => beforeCats.includes(id));
          const serverAppliedChange = required.every(id => nowCats.includes(id));
          if (noChangeNeeded || serverAppliedChange) return {item, duplicate:true, confirmed:true};
        }
      } catch (_) {}

    }
    throw new Error(importCancel ? 'Import canceled' : 'The import could not be confirmed.');

  let chunkUploadsAvailable = false;

  async function uploadFileChunked(file, characterId, categories, sourceUrl='', pending=false) {
    const token = importToken274();
    const chunkSize = 16 * 1024 * 1024;
    const started = performance.now();
    let canceled = false, requestHandle = null, rejectPending = null, cleanupStarted = false;
    const checkCanceled = () => { if (canceled) throw new Error('Import canceled'); };
    const cleanup = () => {
      if (cleanupStarted) return;
      cleanupStarted = true;
      gmRequest({method:'POST', path:`/api/import/chunked/${token}/cancel`,
        headers:{'Content-Type':'application/json'}, data:'{}', timeout:15000})
        .catch(() => console.warn('[NovelAI Media] Temporary upload cleanup deferred until companion restart.'));
    };
    const control = {abort() {
      if (canceled) return;
      canceled = true;
      rejectPending?.(new Error('Import canceled'));
      try { requestHandle?.abort?.(); } catch (_) {}
      cleanup();
    }};
    activeUploadHandle = control;
    const progress = message => {
      if (canceled || activeUploadHandle !== control) return;
      let el = modalRoot.querySelector('#naiUploadProgress253');
      if (!el) {
        const side = modalRoot.querySelector('.reviewSide');
        if (side) {
          el = document.createElement('div'); el.id = 'naiUploadProgress253';
          el.className = 'muted'; el.setAttribute('role','status');
          el.style.marginTop = '10px'; side.appendChild(el);
        } else el = modalRoot.querySelector('#importStatus');
      }
      if (el) el.textContent = message;
    };
    const request = (path, body, timeout=30000) => new Promise((resolve, reject) => {
      checkCanceled();
      let settled = false;
      const finish = (err, value) => {
        if (settled) return;
        settled = true; clearTimeout(timer); rejectPending = null;
        if (err) reject(err); else resolve(value);
      };
      const timer = setTimeout(() => {
        const error = new Error('The import stopped responding. You can retry Save or use X to skip this file.');
        error.retryable = true;
        finish(error);
        try { requestHandle?.abort?.(); } catch (_) {}
      }, timeout);
      rejectPending = err => finish(err);
      const fail = message => {
        const error = new Error(message); error.retryable = true; finish(error);
      };
      try {
        requestHandle = GM_xmlhttpRequest({method:body === null ? 'GET' : 'POST', url:`${API}${path}`,
          headers:{Origin:'https://novelai.net', 'Content-Type':'application/json'},
          data:body === null ? null : JSON.stringify(body), timeout,
          onload:res => {
            let value;
            try { value = JSON.parse(res.responseText || '{}'); }
            catch (_) { fail('The companion returned an invalid upload response.'); return; }
            if (res.status >= 200 && res.status < 300) finish(null, value);
            else {
              const error = new Error(value.error || `Upload failed (${res.status})`);
              error.retryable = res.status === 0 || res.status >= 500;
              finish(error);
            }
          },
          onerror:()=>fail('Could not reach the local companion during import.'),
          ontimeout:()=>fail('The upload step timed out. Retry Save or use X to skip this file.'),
          onabort:()=>finish(new Error('Import canceled')),
        });
      } catch (err) { finish(err); }
    });
    const post = async (action, body, timeout=30000) => {
      for (let attempt=0; ; attempt++) {
        checkCanceled();
        try { return await request(`/api/import/chunked/${token}/${action}`, body, timeout); }
        catch (err) { if (canceled || !err.retryable || attempt >= 1) throw err; }
      }
    };
    try {
      progress('Preparing upload…');
      if (!chunkUploadsAvailable) {
        const health = await request('/api/health', null);
        checkCanceled();
        if (!health?.chunked_uploads) throw new Error('Restart the companion to install the update for larger file imports, then retry Save.');
        chunkUploadsAvailable = true;
      }
      const ready = await post('start', {character_id:characterId, filename:file.name || 'upload',
        size:file.size, content_type:file.type || '', categories:Array.isArray(categories) ? categories : [], source_url:sourceUrl,pending_review:pending});
      let offset = ready.offset;
      if (!Number.isSafeInteger(offset) || offset < 0 || offset > file.size) throw new Error('Invalid upload position from companion.');
      while (offset < file.size) {
        checkCanceled();
        progress(`Saving ${formatBytes(offset)} / ${formatBytes(file.size)} (${Math.floor(offset/file.size*100)}%)…`);
        const bytes = new Uint8Array(await file.slice(offset, offset + chunkSize).arrayBuffer());
        checkCanceled();
        const digest = await crypto.subtle.digest('SHA-256', bytes);
        checkCanceled();
        const sha256 = [...new Uint8Array(digest)].map(x => x.toString(16).padStart(2,'0')).join('');
        const pieces = [];
        for (let i=0; i<bytes.length; i+=32768) pieces.push(String.fromCharCode(...bytes.subarray(i,i+32768)));
        const data = btoa(pieces.join(''));
        const result = await post('chunk', {offset, data, sha256});
        if (result.offset !== offset + bytes.length) throw new Error('Companion acknowledged an unexpected upload position.');
        offset = result.offset;
      }
      checkCanceled();
      progress('Upload complete. Checking and saving file…');
      const result = await post('finish', {}, 120000);
      checkCanceled();
      if (!result?.item?.id) throw new Error('The companion did not confirm the saved file.');
      console.info(`[NovelAI Media timing] chunked import response: ${Math.round(performance.now()-started)} ms`);
      progress('File saved. Updating categories…');
      return result;
    } catch (err) {
      cleanup();
      progress(err?.message || 'Import failed.');
      throw err;
    } finally {
      if (activeUploadHandle === control) activeUploadHandle = null;
    }
  }

  async function uploadFileResilient(file, characterId, categories, sourceUrl='') {
    if (file.size > 8 * 1024 * 1024) return uploadFileChunked(file, characterId, categories, sourceUrl);
    // Small files retain the single-request path; larger files must stay below the
    // browser extension message cap by using bounded, ASCII-encoded chunks.
    const form = new FormData();
    form.append('character_id', characterId);
    form.append('categories', JSON.stringify(Array.isArray(categories) ? categories : []));
    form.append('file', file, file.name || 'upload');
    if (sourceUrl) form.append('source_url', sourceUrl);
    const started = performance.now();
    return await new Promise((resolve, reject) => {
      activeUploadHandle = GM_xmlhttpRequest({
        method:'POST',
        url:`${API}/api/import/file`,
        data:form,
        // A stuck companion request must not permanently lock Rapid Review.  Users can
        // also close a save in progress, which aborts this request and restores controls.
        timeout:90000,
        headers:{Origin:'https://novelai.net'},
        onload:res => {
          activeUploadHandle = null;
          const elapsed = Math.round(performance.now() - started);
          console.info(`[NovelAI Media timing] local import response: ${elapsed} ms`);
          if (res.status >= 200 && res.status < 300) {
            try { resolve(JSON.parse(res.responseText || '{}')); }
            catch (_) { resolve(null); }
            return;
          }
          let message = `Request failed (${res.status})`;
          try { message = JSON.parse(res.responseText || '{}').error || message; } catch (_) {}
          reject(new Error(message));
        },
        onerror:()=>{
          activeUploadHandle = null;
          reject(new Error(importCancel ? 'Import canceled' : 'Could not connect to the local NovelAI Media Library program.'));
        },
        ontimeout:()=>{
          activeUploadHandle = null;
          reject(new Error(importCancel ? 'Import canceled' : 'The local media request timed out.'));
        },
        onabort:()=>{
          activeUploadHandle = null;
          reject(new Error('Import canceled'));
        },
      });
    });
  }

          const duplicate = beforeIds.has(item.id);
          if (!duplicate) return {item, duplicate:false, confirmed:true};
          if (item.media_type === 'video') return {item, duplicate:true, confirmed:true};
          const required = Array.isArray(categories) ? categories : [];
          const beforeCats = Array.isArray(beforeItem?.categories) ? beforeItem.categories : [];
          const nowCats = Array.isArray(item.categories) ? item.categories : [];
          const noChangeNeeded = required.every(id => beforeCats.includes(id));
          const serverAppliedChange = required.every(id => nowCats.includes(id));
          if (noChangeNeeded || serverAppliedChange) return {item, duplicate:true, confirmed:true};
        }
      } catch (_) {}
      await sleep(600);
    }
    return null;
  }

  async function uploadUrlResilient(url, characterId, categories) {
    const before = library || await refreshLibrary();
    const beforeIds = new Set((before.media || []).map(m => m.id));
    const beforeItem = (before.media || []).find(m =>
      m.character_id === characterId && m.source?.kind === 'url' && m.source?.url === url
    ) || null;
    let requestReject = null;
    const requestPromise = new Promise((resolve, reject) => {
      activeUploadHandle = GM_xmlhttpRequest({
        method:'POST', url:`${API}/api/import/url`, timeout:300000,
        headers:{Origin:'https://novelai.net','Content-Type':'application/json'},
        data:JSON.stringify({character_id:characterId, categories, url}),
        onload:res => {
          activeUploadHandle = null;
          if (res.status >= 200 && res.status < 300) {
            try { resolve(JSON.parse(res.responseText || '{}')); } catch (_) { resolve(null); }
          } else {
            let message = `Request failed (${res.status})`;
            try { message = JSON.parse(res.responseText || '{}').error || message; } catch (_) {}
            reject(new Error(message));
          }
        },
        onerror:()=>{ activeUploadHandle = null; reject(new Error(importCancel ? 'Import canceled' : 'Could not connect to the local NovelAI Media Library program.')); },
        ontimeout:()=>{ activeUploadHandle = null; reject(new Error(importCancel ? 'Import canceled' : 'The local media request timed out.')); },
      });
    }).catch(err => { requestReject = err; return null; });
    const confirmPromise = confirmImportedUrl(characterId, url, beforeIds, categories, beforeItem).catch(() => null);
    while (!importCancel) {
      const winner = await Promise.race([
        requestPromise.then(value => ({source:'request', value})),
        confirmPromise.then(value => ({source:'confirm', value})),
      ]);
      if (winner.value?.item?.id) {
        if (winner.source === 'confirm' && activeUploadHandle?.abort) {
          try { activeUploadHandle.abort(); } catch (_) {}
          activeUploadHandle = null;
        }
        return winner.value;
      }
      const other = winner.source === 'request' ? await confirmPromise : await requestPromise;
      if (other?.item?.id) return other;
      break;
    }
    if (importCancel) throw new Error('Import canceled');
    throw requestReject || new Error('The URL import could not be confirmed.');
  }

  function orderedReviewFiles(files) {
    // Use descending natural filename order so numbered batches remain intuitive:
    // Image_1000, Image_999, Image_998 (rather than lexicographic 998, 999, 1000).
    // Array#sort is stable in supported browsers, so identical names retain picker order.
    return [...files].sort((a, b) => String(b?.name || '').localeCompare(String(a?.name || ''), undefined, {numeric:true, sensitivity:'base'}));
  }

  async function downloadUrlPreview(url) {
    return new Promise((resolve, reject) => {
      activeUploadHandle = GM_xmlhttpRequest({
        method:'POST', url:`${API}/api/preview/url`, responseType:'blob', timeout:300000,
        headers:{Origin:'https://novelai.net','Content-Type':'application/json'}, data:JSON.stringify({url}),
        onload:async res => {
          activeUploadHandle = null;
          if (res.status >= 200 && res.status < 300) {
            const header = /(?:^|\n)x-novelai-filename:\s*([^\r\n]+)/i.exec(res.responseHeaders || '');
            const name = header ? decodeURIComponent(header[1].trim()) : 'download';
            resolve(new File([res.response], name, {type:res.response.type}));
          } else {
            let message = res.status === 404 ? 'URL previews need companion runtime 2.7.9. Let the running companion update, then try again.' : `Preview failed (${res.status})`;
            try { message = JSON.parse(await res.response.text()).error || message; } catch (_) {}
            reject(new Error(message));
          }
        },
        onerror:()=>{ activeUploadHandle = null; reject(new Error('Could not download this URL preview.')); },
        ontimeout:()=>{ activeUploadHandle = null; reject(new Error('URL preview timed out.')); },
        onabort:()=>{ activeUploadHandle = null; reject(new Error('Preview canceled.')); },
      });
    });
  }

  async function assignReviewMediaToSet(selectedIds, setId, characterId) {
    if (!reviewSession || !setId) return;
    if(pendingMode()){await resolvePending('set',selectedIds,{set_id:setId});return;}
    const beforeIds = [...reviewSession.ids];
    const beforeIndex = reviewSession.index;
    const selected = new Set(selectedIds);
    const savedIds = [];
    for (const queueId of selectedIds) {
      const m = mediaById(queueId);
      if (!m) continue;
      Object.assign(m, reviewSession.choices?.get(queueId) || {});
      m.in_review=false;
      const persistedId = await persistReviewMedia(m);
      if (!persistedId) throw new Error('The media item could not be saved. Please try again.');
      savedIds.push(persistedId);
    }
    if (!savedIds.length) return;
    await applyReviewSet({id:setId, name:''}, savedIds, characterId);

    // Set assignment is an explicit save action: release draft URLs and remove every
    // assigned image from the review queue, leaving only unchecked items to review.
    for (const queueId of selected) {
      const draft = drafts.get(queueId);
      if (draft) {
        try { URL.revokeObjectURL(draft.url); } catch (_) {}
        drafts.delete(queueId);
      }
      reviewSession.choices?.delete(queueId);
      reviewSession.saved.delete(queueId);reviewSession.selected?.delete(queueId);
    }
    const removedBefore = beforeIds.slice(0, beforeIndex).filter(id => selected.has(id)).length;
    reviewSession.ids = beforeIds.filter(id => !selected.has(id));
    reviewSession.index = Math.max(0, Math.min(beforeIndex - removedBefore, reviewSession.ids.length - 1));

  async function reviewMove(delta) {
    if (!reviewSession || reviewSession.busy) return;
    const session = reviewSession;
    session.busy = true;
    modalRoot.querySelectorAll('button,input').forEach(el => { if (el.id !== 'naiReviewClose270') el.disabled = true; });
    try {
      if(pendingMode()){
        await resolvePending('save',[session.ids[session.index]]);session.busy=false;
        if(!session.ids.length)return finishReviewQueue();await renderReviewQueue();return;
      }
      await saveCurrentReview();
      if (session.canceled || reviewSession !== session) return;
      const next = session.index + delta;
      session.busy = false;
      if (next >= session.ids.length) {
        if (session.ids.length === 1) return finishReviewQueue();
        session.index = 0;
      } else session.index = Math.max(0, next);
      await renderReviewQueue();
    } catch (err) {
      if (!session.canceled) {
        alert(err?.message || String(err));
        modalRoot.querySelectorAll('button,input').forEach(el => el.disabled = false);
      }
    } finally { if (!session.canceled) session.busy = false; }
  }

  async function cancelBusyReview() {
    const session = reviewSession;
    if (!session) return;
    session.canceled = true;
    try { activeUploadHandle?.abort?.(); } catch (_) {}
    activeUploadHandle = null;
    // Discard only the interrupted file; its remaining queue previews stay open.
    const failedId = session.ids[session.index];
    const draft = drafts.get(failedId);
    if (draft) {
      try { URL.revokeObjectURL(draft.url); } catch (_) {}
      drafts.delete(failedId);
    }
    const nextIds = session.ids.filter(id => id !== failedId);
    if (!nextIds.length) {
      session.busy = false;
      return finishReviewQueue(true);
    }
    reviewSession = {
      ...session,
      ids:nextIds,
      index:Math.min(session.index, nextIds.length - 1),
      saved:new Set([...session.saved].filter(id => id !== failedId)),
      choices:new Map([...session.choices].filter(([id]) => id !== failedId)),
      busy:false,
      canceled:false,
    };
    await renderReviewQueue();
  }

  function showReviewQuality(m, element) {
    const draft=drafts.get(m.id);
    const fileBytes=draft?.file?.size ?? m.file_bytes;
    const write = (width,height) => {
      if (Number(width)>0 && Number(height)>0) {
        m.width=Number(width); m.height=Number(height);
        setReviewQuality(m.id,`Source quality: ${width} × ${height} · ${formatBytes(fileBytes)}`);
      } else if (fileBytes != null) setReviewQuality(m.id,`Source quality: ${formatBytes(fileBytes)} · Dimensions unavailable`);
    };
    if (m.width && m.height) return write(m.width,m.height);
    if (element?.tagName === 'IMG') {
      if (element.complete) write(element.naturalWidth,element.naturalHeight);
      else element.addEventListener('load',()=>write(element.naturalWidth,element.naturalHeight),{once:true});
    } else if (element?.tagName === 'VIDEO') {
      element.addEventListener('loadedmetadata',()=>write(element.videoWidth,element.videoHeight),{once:true});
    }
    if (!draft) gmRequest({path:`/api/media/${encodeURIComponent(m.id)}/quality`,timeout:15000}).then(info=>{
      if (!info) return;
      Object.assign(m,info);
      write(info.width,info.height);
    }).catch(()=>{ if (fileBytes != null) setReviewQuality(m.id,`Source quality: ${formatBytes(fileBytes)}`); });
  }

  function decorateEditModal() {
    const deleteBtn = modalRoot.querySelector('#deleteCurrentImage');
    if (!deleteBtn || modalRoot.querySelector('#naiMoveReview270')) return;
    const m = mediaById(activeMediaId());
    if (!m) return;
    const button = document.createElement('button');
    button.type = 'button'; button.id = 'naiMoveReview270';
    button.textContent = m.in_review ? 'Already in Review' : 'Move to Review';
    button.disabled = !!m.in_review;
    deleteBtn.parentElement?.insertBefore(button, deleteBtn);
    button.addEventListener('click', async () => {
      try {
        button.disabled = true; button.textContent = 'Moving…';
        const updated = await gmRequest({method:'POST', path:`/api/media/${encodeURIComponent(m.id)}/review`, headers:{'Content-Type':'application/json'}, data:JSON.stringify({in_review:true})});
        Object.assign(m, updated);
        modalRoot.innerHTML = '';
        scheduleMainFilter(); syncCategoryCounts();
      } catch (err) { button.disabled = false; button.textContent = 'Move to Review'; alert(err?.message || String(err)); }
    });
  }
