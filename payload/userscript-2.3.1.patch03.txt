
// NovelAI Media Library v2.3.1 UI/thumbnail/fullscreen refinement
(() => {
  'use strict';

  const API = 'http://127.0.0.1:8765';
  const host = document.getElementById('nai-media-host');
  const root = host?.shadowRoot;
  if (!root || document.documentElement.dataset.naiMediaV231Patch === '1') return;
  document.documentElement.dataset.naiMediaV231Patch = '1';

  const style = document.createElement('style');
  style.id = 'nai-media-v231-patch';
  style.textContent = `
    :host { --nai-right-edge: 74vw !important; }

    .viewer { position:relative !important; }
    .stages.split { position:absolute !important; inset:0 !important; gap:2px !important; z-index:1 !important; }
    .stages.split .stage { flex:1 1 50% !important; min-height:0 !important; }
    .stages.split ~ .viewerNav {
      position:absolute !important; left:50% !important; bottom:5px !important;
      transform:translateX(-50%) !important; z-index:5 !important;
      padding:4px 7px !important; border-radius:10px !important;
      background:rgba(11,13,29,.80) !important; backdrop-filter:blur(4px) !important;
    }
    .stages.split ~ .badgeRow,
    .stages.split ~ .viewerMeta,
    .stages.split ~ .videoTools.visible { display:none !important; }

    .tile.nai-video-has-thumb .videoMark {
      inset:auto !important; left:6px !important; right:auto !important; top:auto !important; bottom:6px !important;
      display:block !important; width:auto !important; height:auto !important;
      padding:0 !important; background:transparent !important;
      pointer-events:none !important;
    }
    .tile.nai-video-has-thumb .videoMark > * {
      position:static !important; display:block !important;
      padding:2px 6px !important; border-radius:999px !important;
      background:rgba(0,0,0,.68) !important; color:#fff !important;
      font:700 10px system-ui !important;
    }

    .lightbox.nai-split-fullscreen {
      display:grid !important; grid-template-columns:repeat(2,minmax(0,1fr)) !important;
      gap:14px !important; align-items:center !important; justify-items:center !important;
      padding:18px !important;
    }
    .lightbox.nai-split-fullscreen.nai-one-pane { grid-template-columns:minmax(0,1fr) !important; }
    .nai-split-fullscreen-pane {
      width:100% !important; height:calc(100vh - 36px) !important; min-width:0 !important; min-height:0 !important;
      display:flex !important; align-items:center !important; justify-content:center !important; overflow:hidden !important;
    }
    .nai-split-fullscreen-pane img,
    .nai-split-fullscreen-pane video {
      width:100% !important; height:100% !important; max-width:100% !important; max-height:100% !important;
      object-fit:contain !important;
    }
  `;
  root.appendChild(style);

  function gmRequest({ method = 'GET', path, data = null, headers = {}, responseType = 'text', timeout = 300000 }) {
    return new Promise((resolve, reject) => {
      GM_xmlhttpRequest({
        method,
        url: `${API}${path}`,
        data,
        headers: { Origin: 'https://novelai.net', ...headers },
        responseType,
        timeout,
        onload: res => {
          if (res.status >= 200 && res.status < 300) {
            if (responseType === 'blob') return resolve(res.response);
            const text = res.responseText || '';
            if (!text) return resolve(null);
            try { resolve(JSON.parse(text)); } catch (_) { resolve(text); }
            return;
          }
          let message = `Request failed (${res.status})`;
          try { message = JSON.parse(res.responseText || '{}').error || message; } catch (_) {}
          reject(new Error(message));
        },
        onerror: () => reject(new Error('Could not connect to the local NovelAI Media Library program.')),
        ontimeout: () => reject(new Error('The local media request timed out.')),
      });
    });
  }

  const thumbUrls = new Map();

  function mediaIdFromActiveVideo() {
    const video = root.querySelector('.stage.slot-active video') || root.querySelector('.stage:not(.hidden) video');
    if (video) {
      const source = String(video.currentSrc || video.src || '');
      const match = source.match(/\/api\/media\/([0-9a-f]+)\/original(?:$|[?#])/i);
      if (match?.[1]) return match[1];
    }
    return root.querySelector('.tile.active')?.dataset?.id || null;
  }

  function badgeOnly(tile) {
    if (!tile) return;
    const img = tile.querySelector('img[data-mid]');
    const mark = tile.querySelector('.videoMark');
    if (!img || !mark) return;
    tile.classList.add('nai-video-has-thumb');
    mark.innerHTML = '<div>▶ VIDEO</div>';
  }

  function fixVideoTiles() {
    root.querySelectorAll('.tile').forEach(tile => badgeOnly(tile));
  }

  async function ensureVisibleThumbnail(mediaId) {
    if (!mediaId) return;
    const tile = root.querySelector(`.tile[data-id="${mediaId}"]`);
    if (!tile) return;
    let img = tile.querySelector('img[data-mid]');
    try {
      const blob = await gmRequest({ path: `/api/media/${mediaId}/thumb`, responseType: 'blob' });
      if (!blob || !String(blob.type || '').startsWith('image/')) return;
      const oldUrl = thumbUrls.get(mediaId);
      if (oldUrl) URL.revokeObjectURL(oldUrl);
      const url = URL.createObjectURL(blob);
      thumbUrls.set(mediaId, url);
      if (!img) {
        img = document.createElement('img');
        img.loading = 'lazy';
        img.dataset.mid = mediaId;
        tile.insertBefore(img, tile.firstChild);
      }
      img.src = url;
      badgeOnly(tile);
    } catch (_) {}
  }

  const grid = root.getElementById('grid');
  if (grid) {
    const gridObserver = new MutationObserver(() => fixVideoTiles());
    gridObserver.observe(grid, { childList:true, subtree:true });
    fixVideoTiles();
  }

  const toolObserver = new MutationObserver(() => {
    const button = root.getElementById('setThumbBtn');
    if (button?.textContent?.trim() === 'Thumbnail saved') {
      ensureVisibleThumbnail(mediaIdFromActiveVideo());
    }
  });
  toolObserver.observe(root, { childList:true, subtree:true, characterData:true });

  function splitIsActive() {
    return root.getElementById('stages')?.classList.contains('split');
  }

  function makeFullscreenMedia(sourceEl) {
    if (!sourceEl) return null;
    if (sourceEl.tagName === 'IMG') {
      const img = document.createElement('img');
      img.src = sourceEl.currentSrc || sourceEl.src;
      img.alt = sourceEl.alt || '';
      return img;
    }
    if (sourceEl.tagName === 'VIDEO') {
      const video = document.createElement('video');
      video.controls = true;
      video.playsInline = true;
      video.loop = true;
      video.volume = Number.isFinite(sourceEl.volume) ? sourceEl.volume : 0.08;
      video.muted = sourceEl.muted;
      const startTime = Number.isFinite(sourceEl.currentTime) ? sourceEl.currentTime : 0;
      const shouldPlay = !sourceEl.paused;
      video.src = sourceEl.currentSrc || sourceEl.src;
      video.addEventListener('loadedmetadata', () => {
        try {
          const max = Number.isFinite(video.duration) && video.duration > 0 ? Math.max(0, video.duration - 0.05) : startTime;
          video.currentTime = Math.min(startTime, max);
        } catch (_) {}
        if (shouldPlay) video.play().catch(() => {});
      }, { once:true });
      return video;
    }
    return null;
  }

  function closeSplitFullscreen() {
    const lightRoot = root.getElementById('lightRoot');
    if (!lightRoot) return;
    lightRoot.querySelectorAll('video').forEach(v => { try { v.pause(); } catch (_) {} });
    lightRoot.innerHTML = '';
  }

  function openSplitFullscreen() {
    const lightRoot = root.getElementById('lightRoot');
    if (!lightRoot) return;
    const sources = [root.getElementById('stage1'), root.getElementById('stage2')]
      .map(stage => stage?.querySelector('img,video'))
      .filter(Boolean);
    if (!sources.length) return;

    lightRoot.innerHTML = '';
    const box = document.createElement('div');
    box.className = `lightbox nai-split-fullscreen${sources.length === 1 ? ' nai-one-pane' : ''}`;
    const close = document.createElement('button');
    close.className = 'lightClose';
    close.textContent = '×';
    box.appendChild(close);

    for (const source of sources) {
      const pane = document.createElement('div');
      pane.className = 'nai-split-fullscreen-pane';
      const media = makeFullscreenMedia(source);
      if (media) pane.appendChild(media);
      box.appendChild(pane);
    }

    lightRoot.appendChild(box);
    close.addEventListener('click', closeSplitFullscreen);
    box.addEventListener('click', event => { if (event.target === box) closeSplitFullscreen(); });
  }

  document.addEventListener('click', event => {
    const path = event.composedPath?.() || [];
    if (!path.includes(host)) return;

    const seekButton = path.find(node => node?.getAttribute?.('data-seek-marker') !== null && node?.hasAttribute?.('data-seek-marker'));
    if (seekButton) {
      const video = root.querySelector('.stage.slot-active video') || root.querySelector('.stage:not(.hidden) video');
      if (video) {
        const wasPaused = video.paused;
        setTimeout(() => {
          if (!video.isConnected) return;
          if (wasPaused) video.pause();
          else video.play().catch(() => {});
        }, 30);
      }
      return;
    }

    if (!splitIsActive()) return;
    const fullscreenButton = path.find(node => node?.id === 'fullBtn');
    const stageImage = path.find(node => node?.tagName === 'IMG' && node?.closest?.('.stage'));
    if (!fullscreenButton && !stageImage) return;

    event.preventDefault();
    event.stopImmediatePropagation();
    openSplitFullscreen();
  }, true);

  document.addEventListener('keydown', event => {
    if (event.key === 'Escape' && root.querySelector('.nai-split-fullscreen')) closeSplitFullscreen();
  }, true);

  window.addEventListener('beforeunload', () => {
    for (const url of thumbUrls.values()) URL.revokeObjectURL(url);
    thumbUrls.clear();
  }, { once:true });
})();
