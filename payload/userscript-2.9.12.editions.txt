function drawCentered(ctx,text,x,y,size,family,colors,maxWidth,maxHeight){

 const scale=Math.min(1,maxWidth?maxWidth/inkWidth:1,maxHeight?maxHeight/inkHeight:1);

 const baseline=y+(asc-desc)/2,origin=x-(m.actualBoundingBoxRight-m.actualBoundingBoxLeft)/2;
 const fill=ctx.createLinearGradient(0,y-asc/2,0,y+asc/2+desc);
 fill.addColorStop(0,colors[0]);fill.addColorStop(.27,colors[0]);fill.addColorStop(.48,colors[1]);fill.addColorStop(.52,colors[2]);fill.addColorStop(.59,colors[0]);fill.addColorStop(1,colors[1]);

}
