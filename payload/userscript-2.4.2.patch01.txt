  function uiState() {
    try { return JSON.parse(localStorage.getItem(UI_STATE_KEY) || '{}') || {}; }
    catch (_) { return {}; }
  }

  function activeSlot() {
    return stageEls[1]?.classList.contains('slot-active') ? 1 : 0;
  }

  function slotMediaId(slot) {
    const state = uiState();
    const ids = Array.isArray(state.slotMediaIds) ? state.slotMediaIds : [];
    return ids[slot] || (slot === Number(state.activeSlot || 0) ? state.selectedMediaId : null) || null;
  }

  function restoreVideoTools() {
    videoTools.classList.remove('naiVideoOverlayTools');
    videoTools.style.removeProperty('top');
    videoTools.style.removeProperty('left');
    videoTools.style.removeProperty('max-width');
    if (!originalToolsParent || videoTools.parentNode === originalToolsParent) return;
    if (originalToolsNext && originalToolsNext.parentNode === originalToolsParent) {
      originalToolsParent.insertBefore(videoTools, originalToolsNext);
    } else {
      originalToolsParent.appendChild(videoTools);
    }
  }

  function visibleVideoRect(video, container) {
    const cw = Math.max(1, container.clientWidth || 1);
    const ch = Math.max(1, container.clientHeight || 1);
    const vw = Number(video.videoWidth || 0);
    const vh = Number(video.videoHeight || 0);
    if (!vw || !vh) return { left:8, top:8, width:Math.max(80, cw - 16), height:Math.max(1, ch - 16) };
    const scale = Math.min(cw / vw, ch / vh);
    const width = vw * scale;
    const height = vh * scale;
    return { left:(cw - width) / 2, top:(ch - height) / 2, width, height };
  }

  function placeSplitVideoTools() {
    placeQueued = false;
    const split = stages.classList.contains('split');
    const slot = activeSlot();
    const stage = stageEls[slot];
    const video = stage?.querySelector('video');
    const visible = videoTools.classList.contains('visible');
    if (!split || !stage || !video || !visible) {
      restoreVideoTools();
      return;
    }
    if (videoTools.parentNode !== stage) stage.appendChild(videoTools);
    videoTools.classList.add('naiVideoOverlayTools');
    const r = visibleVideoRect(video, stage);
    videoTools.style.left = `${Math.max(6, r.left + 7)}px`;
    videoTools.style.top = `${Math.max(6, r.top + 7)}px`;
    videoTools.style.maxWidth = `${Math.max(120, r.width - 14)}px`;
  }

  function queueSplitPlacement() {
    if (placeQueued) return;
    placeQueued = true;
    requestAnimationFrame(placeSplitVideoTools);
  }

  new MutationObserver(queueSplitPlacement).observe(videoTools, { childList:true, attributes:true, attributeFilter:['class'] });
  new MutationObserver(queueSplitPlacement).observe(stages, { attributes:true, attributeFilter:['class'] });
  stageEls.forEach(stage => {
    if (stage) new MutationObserver(queueSplitPlacement).observe(stage, { childList:true });
  });
  window.addEventListener('resize', queueSplitPlacement, { passive:true });
  root.addEventListener('click', event => {
    const t = event.target;
    if (t?.closest?.('.tile[data-id]') || t?.id === 'splitBtn' || t?.id === 'naiSlotSelectV239_1' || t?.id === 'naiSlotSelectV239_2') {
      [0,60,180].forEach(ms => setTimeout(queueSplitPlacement, ms));
    }
  });
  stageEls.forEach(stage => stage?.addEventListener('loadedmetadata', queueSplitPlacement, true));

  function gmRequest({ method='GET', path, data=null, headers={}, responseType='text', timeout=300000 }) {
    return new Promise((resolve, reject) => {
      GM_xmlhttpRequest({
        method,
        url:`${API}${path}`,
        data,
        headers:{ Origin:'https://novelai.net', ...headers },
        responseType,
        timeout,
        onload:res => {
          if (res.status >= 200 && res.status < 300) {
            if (responseType === 'blob') return resolve(res.response);
            const text = res.responseText || '';
            if (!text) return resolve(null);
            try { resolve(JSON.parse(text)); } catch (_) { resolve(text); }
            return;
          }
          let message = `Request failed (${res.status})`;
          try { message = JSON.parse(res.responseText || '{}').error || message; } catch (_) {}
          reject(new Error(message));
        },
        onerror:()=>reject(new Error('Could not connect to the local NovelAI Media Library program.')),
        ontimeout:()=>reject(new Error('The local media request timed out.')),
      });
    });
  }

  async function mediaMeta(mediaId, force=false) {
    if (!force && mediaCache.has(mediaId)) return mediaCache.get(mediaId);
    const lib = await gmRequest({ path:'/api/library' });
    for (const m of (lib?.media || [])) if (m?.id) mediaCache.set(m.id, m);
    return mediaCache.get(mediaId) || null;
  }

  function formatTime(seconds) {
    const value = Math.max(0, Number(seconds || 0));
    const mins = Math.floor(value / 60);
    const secs = Math.floor(value % 60).toString().padStart(2, '0');
    return `${mins}:${secs}`;
  }

  function waitFor(video, eventName, timeout=30000) {
    return new Promise((resolve, reject) => {
      let timer = 0;
      const cleanup = () => {
        clearTimeout(timer);
        video.removeEventListener(eventName, done);
        video.removeEventListener('error', failed);
      };
      const done = () => { cleanup(); resolve(); };
      const failed = () => { cleanup(); reject(new Error('Could not load the video frame.')); };
      timer = setTimeout(() => { cleanup(); reject(new Error(`Timed out waiting for video ${eventName}.`)); }, timeout);
      video.addEventListener(eventName, done, { once:true });
      video.addEventListener('error', failed, { once:true });
    });
  }

  async function canvasBlobFromVideo(video) {
    if (!video?.videoWidth || !video?.videoHeight) throw new Error('Video frame is not ready yet.');
    const maxSide = 960;
    const scale = Math.min(1, maxSide / Math.max(video.videoWidth, video.videoHeight));
    const canvas = document.createElement('canvas');
    canvas.width = Math.max(1, Math.round(video.videoWidth * scale));
    canvas.height = Math.max(1, Math.round(video.videoHeight * scale));
    const ctx = canvas.getContext('2d');
    if (!ctx) throw new Error('Could not capture this video frame.');
    ctx.drawImage(video, 0, 0, canvas.width, canvas.height);
    return await new Promise((resolve, reject) => {
      try {
        canvas.toBlob(blob => blob ? resolve(blob) : reject(new Error('Could not create thumbnail image.')), 'image/jpeg', 0.9);
      } catch (err) { reject(err); }
    });
  }

  async function captureFrame(mediaId, seconds, displayedVideo) {
    try {
      if (displayedVideo?.readyState >= 2) return await canvasBlobFromVideo(displayedVideo);
    } catch (_) {}
    const sourceBlob = await gmRequest({ path:`/api/media/${encodeURIComponent(mediaId)}/original`, responseType:'blob' });
    const sourceUrl = URL.createObjectURL(sourceBlob);
    const video = document.createElement('video');
    video.preload = 'auto';
    video.muted = true;
    video.playsInline = true;
    video.src = sourceUrl;
    try {
      if (video.readyState < 1) await waitFor(video, 'loadedmetadata');
      const duration = Number.isFinite(video.duration) ? video.duration : 0;
      const target = duration > 0 ? Math.min(Math.max(0, seconds), Math.max(0, duration - 0.01)) : Math.max(0, seconds);
      if (Math.abs(video.currentTime - target) > 0.001) {
        video.currentTime = target;
        await waitFor(video, 'seeked');
      } else if (video.readyState < 2) {
        await waitFor(video, 'loadeddata');
      }
      return await canvasBlobFromVideo(video);
    } finally {
      try { video.pause(); } catch (_) {}
      video.removeAttribute('src');
      video.load();
      URL.revokeObjectURL(sourceUrl);
    }
  }

  function applyChangedThumb(mediaId) {
    const url = changedThumbUrls.get(mediaId);
    if (!url) return;
    const tile = root.querySelector(`.tile[data-id="${mediaId}"]`);
    if (!tile) return;
    let img = tile.querySelector('img[data-mid]');
    if (!img) {
      img = document.createElement('img');
      img.loading = 'eager';
      img.dataset.mid = mediaId;
      tile.insertBefore(img, tile.firstChild);
    }
    if (img.src !== url) img.src = url;
    img.alt = '';
  }

  function rememberChangedThumb(mediaId, blob) {
    const old = changedThumbUrls.get(mediaId);
    if (old) URL.revokeObjectURL(old);
    changedThumbUrls.set(mediaId, URL.createObjectURL(blob));
    applyChangedThumb(mediaId);
  }

  function mediaIdFromVideo(video) {
    const source = String(video?.currentSrc || video?.src || '');
    const m = source.match(/\/api\/media\/([0-9a-f]+)\/original(?:$|[?#])/i);
    return m?.[1] || null;
  }

  function positionFullscreenToolbar(toolbar, video, container) {
    if (!toolbar?.isConnected || !video?.isConnected || !container?.isConnected) return;
    const cr = container.getBoundingClientRect();
    const vr = video.getBoundingClientRect();
    const left = Math.max(7, vr.left - cr.left + 7);
    let top = Math.max(7, vr.top - cr.top + 7);
    if (container.classList.contains('naiThirdMedia')) top = Math.max(50, top);
    toolbar.style.left = `${left}px`;
    toolbar.style.top = `${top}px`;
    toolbar.style.maxWidth = `${Math.max(140, Math.min(cr.width - left - 7, vr.width - 14))}px`;
  }

  async function renderFullscreenToolbar(toolbar, video, mediaId) {
    if (!toolbar?.isConnected || !video?.isConnected) return;
    let m;
    try { m = await mediaMeta(mediaId); } catch (_) { m = null; }
    const markers = Array.isArray(m?.video_markers) ? [m.video_markers[0] ?? null, m.video_markers[1] ?? null, m.video_markers[2] ?? null] : [null, null, null];
    toolbar.innerHTML = '';

    const thumb = document.createElement('button');
    thumb.type = 'button';
    thumb.textContent = 'Set thumbnail';
    thumb.title = 'Use the current fullscreen video frame as this video thumbnail';
    thumb.addEventListener('click', async event => {
      event.preventDefault();
      event.stopPropagation();
      if (thumb.disabled) return;
      thumb.disabled = true;
      thumb.textContent = 'Saving…';
      try {
        const seconds = Number.isFinite(video.currentTime) ? video.currentTime : 0;
        const blob = await captureFrame(mediaId, seconds, video);
        const updated = await gmRequest({
          method:'POST',
          path:`/api/media/${encodeURIComponent(mediaId)}/thumbnail?seconds=${encodeURIComponent(seconds)}`,
          headers:{'Content-Type':'image/jpeg'},
          data:blob,
        });
        if (updated?.id) mediaCache.set(updated.id, updated);
        rememberChangedThumb(mediaId, blob);
        thumb.textContent = 'Saved';
      } catch (err) {
        alert(err?.message || String(err));
        thumb.textContent = 'Set thumbnail';
      } finally {
        setTimeout(() => {
          if (!thumb.isConnected) return;
          thumb.disabled = false;
          if (thumb.textContent === 'Saved') thumb.textContent = 'Set thumbnail';
        }, 700);
      }
    });
    toolbar.appendChild(thumb);

    markers.forEach((marker, idx) => {
      const group = document.createElement('span');
      group.className = 'naiMarkerGroup';
      const seek = document.createElement('button');
      seek.type = 'button';
      seek.textContent = marker == null ? `T${idx + 1} —` : `T${idx + 1} ${formatTime(marker)}`;
      seek.disabled = marker == null;
      seek.title = marker == null ? `Timestamp ${idx + 1} is not set` : `Jump to timestamp ${idx + 1}`;
      seek.addEventListener('click', event => {
        event.preventDefault();
        event.stopPropagation();
        if (marker == null) return;
        try { video.currentTime = Number(marker); } catch (_) {}
      });
      const set = document.createElement('button');
      set.type = 'button';
      set.className = 'naiMarkerSet';
      set.textContent = `Set T${idx + 1}`;
      set.title = `Save the current video time to timestamp ${idx + 1}`;
      set.addEventListener('click', async event => {
        event.preventDefault();
        event.stopPropagation();
        if (set.disabled) return;
        set.disabled = true;
        const old = set.textContent;
        set.textContent = 'Saving…';
        try {
          const latest = await mediaMeta(mediaId, true);
          const next = Array.isArray(latest?.video_markers) ? [latest.video_markers[0] ?? null, latest.video_markers[1] ?? null, latest.video_markers[2] ?? null] : [null, null, null];
          next[idx] = Number.isFinite(video.currentTime) ? video.currentTime : 0;
          const updated = await gmRequest({
            method:'POST',
            path:`/api/media/${encodeURIComponent(mediaId)}/video-meta`,
            headers:{'Content-Type':'application/json'},
            data:JSON.stringify({markers:next}),
          });
          if (updated?.id) mediaCache.set(updated.id, updated);
          await renderFullscreenToolbar(toolbar, video, mediaId);
        } catch (err) {
          alert(err?.message || String(err));
          set.textContent = old;
          set.disabled = false;
        }
      });
      group.append(seek, set);
      toolbar.appendChild(group);
    });
    positionFullscreenToolbar(toolbar, video, toolbar.parentElement);
  }

  function ensureFullscreenToolbar(container, video, mediaId) {
    if (!container || !video || !mediaId) return;
    let toolbar = container.querySelector(`.naiFullscreenVideoTools[data-media-id="${mediaId}"]`);
    if (!toolbar) {
      toolbar = document.createElement('div');
      toolbar.className = 'naiFullscreenVideoTools';
      toolbar.dataset.mediaId = mediaId;
      container.appendChild(toolbar);
      renderFullscreenToolbar(toolbar, video, mediaId);
      video.addEventListener('loadedmetadata', () => positionFullscreenToolbar(toolbar, video, container));
    } else {
      positionFullscreenToolbar(toolbar, video, container);
    }
  }

  function decorateFullscreen() {
    const box = root.querySelector('.naiTripleFullscreen');
    if (!box) return;
    const panes = Array.from(box.children).filter(el => el.classList?.contains('naiTriplePane'));
    for (let i = 0; i < Math.min(2, panes.length); i++) {
      const video = Array.from(panes[i].children).find(el => el.tagName === 'VIDEO');
      if (!video) continue;
      const id = slotMediaId(i) || mediaIdFromVideo(video);
      if (id) ensureFullscreenToolbar(panes[i], video, id);
    }
    const thirdMedia = box.querySelector('.naiThirdMedia:not([hidden])');
    const thirdVideo = thirdMedia ? Array.from(thirdMedia.children).find(el => el.tagName === 'VIDEO') : null;
    if (thirdMedia && thirdVideo) {
      const id = mediaIdFromVideo(thirdVideo);
      if (id) ensureFullscreenToolbar(thirdMedia, thirdVideo, id);
    }
  }

  function scheduleFullscreenDecorate() {
    clearTimeout(fullscreenDecorateTimer);
    [0,80,220,650].forEach(ms => setTimeout(() => {
      decorateFullscreen();
      root.querySelectorAll('.naiFullscreenVideoTools').forEach(toolbar => {
        const container = toolbar.parentElement;
        const video = container ? Array.from(container.children).find(el => el.tagName === 'VIDEO') : null;
        if (video) positionFullscreenToolbar(toolbar, video, container);
      });
    }, ms));
  }

  new MutationObserver(scheduleFullscreenDecorate).observe(lightRoot, { childList:true });
  root.addEventListener('click', event => {
    const tile = event.target.closest?.('.right.naiTripleBrowserPanel .tile[data-id]');
    if (tile || event.target.closest?.('#fullBtn, .stage img, .naiPaneToggle, .naiThirdChange')) scheduleFullscreenDecorate();
  }, true);
  window.addEventListener('resize', () => {
    decorateFullscreen();
    root.querySelectorAll('.naiFullscreenVideoTools').forEach(toolbar => {
      const container = toolbar.parentElement;
      const video = container ? Array.from(container.children).find(el => el.tagName === 'VIDEO') : null;
      if (video) positionFullscreenToolbar(toolbar, video, container);
    });
  }, { passive:true });

  root.addEventListener('click', () => {
    if (!changedThumbUrls.size) return;
    setTimeout(() => { for (const id of changedThumbUrls.keys()) applyChangedThumb(id); }, 0);
  });

  window.addEventListener('beforeunload', () => {
    for (const url of changedThumbUrls.values()) URL.revokeObjectURL(url);
    changedThumbUrls.clear();
  }, { once:true });

  queueSplitPlacement();
  scheduleFullscreenDecorate();
})();
