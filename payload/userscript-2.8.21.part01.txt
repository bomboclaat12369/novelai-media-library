      GM_xmlhttpRequest({
        method,
        url: `${API}${path}`,
        data,
        headers: { Origin: 'https://novelai.net', ...headers },
        responseType,
        timeout,
        onload: (res) => {
          if (res.status >= 200 && res.status < 300) {
            if (responseType === 'blob') return resolve(res.response);
            const text = res.responseText ?? '';
            if (!text) return resolve(null);
            try { resolve(JSON.parse(text)); } catch (_) { resolve(text); }
          } else {
            let message = `Request failed (${res.status})`;
            try {
              const parsed = JSON.parse(res.responseText || '{}');
              if (parsed.error) message = parsed.error;
            } catch (_) {}
            reject(new Error(message));
          }
        },
        onerror: () => reject(new Error('Could not connect to the local NovelAI Media Library program.')),
        ontimeout: () => reject(new Error('The local media request timed out.')),
      });
    });

  function escapeHtml(s) {
    return String(s ?? '').replace(/[&<>"']/g, c => ({'&':'&amp;','<':'&lt;','>':'&gt;','"':'&quot;',"'":'&#039;'}[c]));
  }

  function currentCharacter() {
    return state.library?.characters?.find(c => c.id === state.characterId) || null;

  function allCurrentMedia() {
    if (!state.library || !state.characterId) return [];
    return state.library.media.filter(m => m.character_id === state.characterId);
  }

  function filteredMedia() {
    let items = allCurrentMedia();
    if (state.categoryId === 'favorites') items = items.filter(m => !!m.favorite && m.media_type !== 'video');
    else if (state.categoryId === 'videos') items = items.filter(m => m.media_type === 'video');
    else if (state.categoryId === 'all') items = items.filter(m => m.media_type !== 'video' && m.in_all !== false);
    else items = items.filter(m => m.media_type !== 'video' && (m.categories || []).includes(state.categoryId));
    const q = state.search.trim().toLowerCase();
    if (q) items = items.filter(m => (m.original_name || '').toLowerCase().includes(q));
    return items;
  }

    if (id === 'videos') return 'Videos';

    return c?.categories?.find(x => x.id === id)?.name || '';

      opt.textContent = c.name;

  }

  function renderCategories() {
    catWrap.innerHTML = '';

    if (!c) return;

      if (catId === 'favorites') return m.media_type !== 'video' && !!m.favorite;
      if (catId === 'videos') return m.media_type === 'video';
      return m.media_type !== 'video' && (m.categories || []).includes(catId);
    }).length;
    const cats = [{ id: 'all', name: 'All' }, { id: 'favorites', name: '★ Favorites' }, { id: 'videos', name: '▶ Videos' }, ...(c.categories || [])];
    if (!cats.some(x => x.id === state.categoryId)) state.categoryId = 'all';
    for (const cat of cats) {
      const b = document.createElement('button');
      b.className = `cat${cat.id === state.categoryId ? ' active' : ''}`;
      b.textContent = `${cat.name} ${countFor(cat.id)}`;

        state.categoryId = cat.id;
        persistUI();
        renderCategories();
        renderGrid();
        updateCounter();
      });
      catWrap.appendChild(b);
    }
  }

  const io = new IntersectionObserver((entries) => {
    for (const entry of entries) {
      if (!entry.isIntersecting) continue;
      const tile = entry.target;
      io.unobserve(tile);
      const img = tile.querySelector('img[data-mid]');
      if (img) loadThumb(img.dataset.mid, img);
    }

  function cacheThumb(id, url) {
    if (state.thumbCache.has(id)) {
      const old = state.thumbCache.get(id);
      state.thumbCache.delete(id);
      state.thumbCache.set(id, old);
      return old;
    }
    state.thumbCache.set(id, url);
    while (state.thumbCache.size > MAX_THUMB_CACHE) {
      const [oldId, oldUrl] = state.thumbCache.entries().next().value;
      state.thumbCache.delete(oldId);
      URL.revokeObjectURL(oldUrl);
    }
    return url;
  }

  async function loadThumb(id, img) {
    try {
      if (state.thumbCache.has(id)) {
        const url = state.thumbCache.get(id);
        state.thumbCache.delete(id);
        state.thumbCache.set(id, url);
        img.src = url;
        return;
      }
      const blob = await gmRequest({ path: `/api/media/${id}/thumb`, responseType: 'blob' });
      const url = cacheThumb(id, URL.createObjectURL(blob));
      img.src = url;
    } catch (_) {
      img.alt = 'Thumbnail unavailable';
    }
  }

  function renderGrid(itemsOverride = null) {
    grid.innerHTML = '';
    if (!state.connected) {
      grid.innerHTML = '<div class="empty">Start the local companion program, then click Refresh.</div>';
      return;
    }
    if (!state.characterId) {
      grid.innerHTML = '<div class="empty">Create a character, then add media.</div>';
      return;
    }
    const items = itemsOverride || filteredMedia();
    if (!items.length) {
      grid.innerHTML = '<div class="empty">No media in this view yet.<br>Use “+ Media” to import local files or URLs.</div>';
      return;
    }
    const frag = document.createDocumentFragment();
    for (const m of items) {
      const tile = document.createElement('div');
      tile.className = `tile${m.id === state.selectedMediaId ? ' active' : ''}`;
      tile.title = m.original_name || '';
      tile.dataset.id = m.id;
      if (m.thumb_rel || (itemsOverride && m.media_type === 'image')) {
        const img = document.createElement('img');
        img.loading = 'lazy';
        img.dataset.mid = m.id;
        tile.appendChild(img);
        io.observe(tile);
      }
      if (m.media_type === 'video') {
        const mark = document.createElement('div');
        mark.className = 'videoMark';
        mark.innerHTML = m.thumb_rel ? `<div style="position:absolute; left:6px; bottom:6px; padding:2px 6px; border-radius:999px; background:rgba(0,0,0,.6)">▶ VIDEO</div>` : `<div>▶ VIDEO</div><div class="videoName">${escapeHtml(m.original_name || '')}</div>`;
        tile.appendChild(mark);
      }
      if (m.favorite) {
        const star = document.createElement('div');
        star.className = 'star';
        star.textContent = '★';

  function formatBytes(value) {
    const bytes = Number(value);
    if (!Number.isFinite(bytes) || bytes < 0) return '';
    if (bytes < 1024) return `${Math.round(bytes)} B`;
    if (bytes < 1024 ** 2) return `${(bytes / 1024).toFixed(1)} KB`;
    if (bytes < 1024 ** 3) return `${(bytes / 1024 ** 2).toFixed(1)} MB`;
    return `${(bytes / 1024 ** 3).toFixed(2)} GB`;
  }

  async function loadMediaQuality(m) {
    const token = ++state.qualityToken;
    qualityInfo.textContent = 'Source quality: loading…';
    try {
      const info = await gmRequest({path:`/api/media/${encodeURIComponent(m.id)}/quality`, timeout:15000});
      if (token !== state.qualityToken || activeMedia()?.id !== m.id) return;
      const size = formatBytes(info?.file_bytes);
      const dimensions = Number(info?.width) > 0 && Number(info?.height) > 0 ? `${info.width} × ${info.height}` : '';
      const details = [dimensions, size].filter(Boolean).join(' · ');
      qualityInfo.textContent = details ? `Source quality: ${details}` : 'Source quality: unavailable';
    } catch (_) {
      if (token === state.qualityToken && activeMedia()?.id === m.id) qualityInfo.textContent = 'Source quality: unavailable';
    }
  }

  function formatTime(seconds) {
    if (seconds === null || seconds === undefined || Number.isNaN(Number(seconds))) return '—';
    const total = Math.max(0, Math.floor(Number(seconds)));
    const h = Math.floor(total / 3600);
    const m = Math.floor((total % 3600) / 60);
    const s = total % 60;
    return h > 0 ? `${h}:${String(m).padStart(2,'0')}:${String(s).padStart(2,'0')}` : `${m}:${String(s).padStart(2,'0')}`;
  }

  function currentVideoTime() {
    const v = activeVideoElement();
    return v && Number.isFinite(v.currentTime) ? Math.max(0, v.currentTime) : 0;
  }

  async function canvasThumbnailBlob(video) {
    if (!video || !video.videoWidth || !video.videoHeight) throw new Error('Wait for the video frame to appear first.');
    const maxSide = 840;
    const scale = Math.min(1, maxSide / Math.max(video.videoWidth, video.videoHeight));
    const canvas = document.createElement('canvas');
    canvas.width = Math.max(1, Math.round(video.videoWidth * scale));
    canvas.height = Math.max(1, Math.round(video.videoHeight * scale));
    const ctx = canvas.getContext('2d', { alpha: false });
    ctx.drawImage(video, 0, 0, canvas.width, canvas.height);
    const blob = await new Promise(resolve => canvas.toBlob(resolve, 'image/jpeg', 0.86));
    if (!blob) throw new Error('Could not capture this video frame.');
    return blob;
  }

  async function setVideoSourceWithFallback(video, m, slot = null) {
    video.controls = true;
    video.preload = 'metadata';
    video.playsInline = true;
    video.loop = true;
    video.volume = 0.08;
    video.src = directOriginalUrl(m.id);
    let triedFallback = false;
    video.addEventListener('error', async () => {
      if (triedFallback || !video.isConnected) return;
      triedFallback = true;
      try {
        const blob = await gmRequest({ path: `/api/media/${m.id}/original`, responseType: 'blob', timeout: 300000 });
        const url = URL.createObjectURL(blob);
        if (slot !== null) {
          revokeSlotUrl(slot);
          state.slotObjectUrls[slot] = url;
        } else {
          video._naiBlobUrl = url;
        }
        video.src = url;
        video.load();
      } catch (_) {}
    }, { once: true });
  }

          const file = replaceInput.files?.[0];
          if (!file) return;
          if (!confirm(`Replace “${m.original_name || 'this image'}” with “${file.name}”?\n\nIts categories, favorite/Review state, position, crop settings, and set membership will be preserved.`)) {
            replaceInput.value = '';
            return;
          }
          replaceButton.disabled = true;
          replaceButton.textContent = 'Replacing…';
          try {
            const form = new FormData();
            form.append('file', file, file.name || 'replacement');
            const updated = await gmRequest({
              method: 'POST',
              path: `/api/media/${encodeURIComponent(m.id)}/replace`,
              data: form,
              timeout: 300000,
            });
            Object.assign(m, updated || {});
            const cached = state.thumbCache.get(m.id);
            if (cached) { URL.revokeObjectURL(cached); state.thumbCache.delete(m.id); }
            close();
            await loadLibrary(true);
            if (state.library.media.some(x => x.id === m.id)) await selectMedia(m.id);
          } catch (err) {
            alert(err.message);
            replaceButton.disabled = false;
            replaceButton.textContent = 'Replace Source Image';
          } finally {
            replaceInput.value = '';
          }
        });
      }
    }

    const deleteCategoryBtn = modalRoot.querySelector('#deleteCategoryBtn');
    if (deleteCategoryBtn) {
      deleteCategoryBtn.addEventListener('click', async () => {
        const select = modalRoot.querySelector('#deleteCategorySelect');
        const categoryId = select?.value;
        const category = realCategories.find(cat => cat.id === categoryId);
        if (!category) return;
        const count = allCurrentMedia().filter(item => (item.categories || []).includes(categoryId)).length;
        if (!confirm(`Delete the category “${category.name}”?\n\nIt will be removed from ${count} image${count === 1 ? '' : 's'}. The images themselves will not be deleted.`)) return;
        try {
          await gmRequest({ method: 'DELETE', path: `/api/categories/${categoryId}` });
          if (state.categoryId === categoryId) state.categoryId = 'all';
          close();
          await loadLibrary();
        } catch (err) {
          alert(`${err.message}\n\nIf you are still running the original companion program, replace companion.pyw with the current companion included with this update and restart it.`);
        }
      });
    }
  }

    modalRoot.innerHTML = `

      </div>`;
    const wrap = modalRoot.querySelector('.modalWrap');

    modalRoot.querySelectorAll('[data-close]').forEach(b => b.addEventListener('click', close));
    wrap.addEventListener('click', e => { if (e.target === wrap) close(); });
    let activeTab = 'local';
    modalRoot.querySelectorAll('.tab').forEach(b => b.addEventListener('click', () => {
      activeTab = b.dataset.tab;
      modalRoot.querySelectorAll('.tab').forEach(x => x.classList.toggle('active', x === b));
      $('#noop');
      modalRoot.querySelector('#localPane').classList.toggle('hidden', activeTab !== 'local');
      modalRoot.querySelector('#urlPane').classList.toggle('hidden', activeTab !== 'url');
    }));
    modalRoot.querySelector('#modalAddCategory').addEventListener('click', async () => {
      const cat = await createCategoryFlow();
      if (!cat || !modalRoot.childElementCount) return;
      modalRoot.querySelector('#importCats').innerHTML = categoryCheckboxes([]);
    });
    modalRoot.querySelector('#modalEditImages').addEventListener('click', () => {
      close();
      openEditImagesModal();
    });

    let pendingLocalFiles = [];
    const fileInput = modalRoot.querySelector('#fileInput');
    const dropZone = modalRoot.querySelector('#fileDropZone');
    const fileSummary = modalRoot.querySelector('#fileSummary');
    const clearFiles = modalRoot.querySelector('#clearFiles');

    function isLikelyMediaFile(file) {
      if (!file) return false;
      if ((file.type || '').startsWith('image/') || (file.type || '').startsWith('video/')) return true;
      return /\.(?:jpe?g|png|gif|webp|bmp|tiff?|avif|heic|heif|mp4|webm|mov|m4v|avi|mkv)$/i.test(file.name || '');
    }

    function addPendingFiles(files) {
      const incoming = [...files].filter(isLikelyMediaFile);
      const seen = new Set(pendingLocalFiles.map(f => `${f.name}::${f.size}::${f.lastModified}`));
      for (const file of incoming) {
        const key = `${file.name}::${file.size}::${file.lastModified}`;
        if (!seen.has(key)) {
          pendingLocalFiles.push(file);
          seen.add(key);
        }
      }
      paintFileSummary();
    }

    function paintFileSummary() {
      if (!pendingLocalFiles.length) {
        fileSummary.classList.add('hidden');
        fileSummary.innerHTML = '';
        clearFiles.disabled = true;
        return;
      }
      const shown = pendingLocalFiles.slice(0, 6).map(f => escapeHtml(f.name));
      const more = pendingLocalFiles.length > shown.length ? `<br>…and ${pendingLocalFiles.length - shown.length} more` : '';
      fileSummary.innerHTML = `<strong>${pendingLocalFiles.length} file${pendingLocalFiles.length === 1 ? '' : 's'} selected</strong><br>${shown.join('<br>')}${more}`;
      fileSummary.classList.remove('hidden');
      clearFiles.disabled = false;
    }

    function chooseFiles() { fileInput.click(); }
    dropZone.addEventListener('click', chooseFiles);
    dropZone.addEventListener('keydown', e => {
      if (e.key === 'Enter' || e.key === ' ') { e.preventDefault(); chooseFiles(); }
    });
    modalRoot.querySelector('#chooseMoreFiles').addEventListener('click', chooseFiles);
    fileInput.addEventListener('change', () => {
      addPendingFiles(fileInput.files || []);
      fileInput.value = '';
    });
    clearFiles.addEventListener('click', () => {
      pendingLocalFiles = [];
      fileInput.value = '';
      paintFileSummary();
    });
    for (const eventName of ['dragenter','dragover']) {
      dropZone.addEventListener(eventName, e => {
        e.preventDefault();
        e.stopPropagation();
        if (e.dataTransfer) e.dataTransfer.dropEffect = 'copy';
        dropZone.classList.add('dragover');
      });
    }
    for (const eventName of ['dragleave','drop']) {
      dropZone.addEventListener(eventName, e => {
        e.preventDefault();
        e.stopPropagation();
        dropZone.classList.remove('dragover');
      });
    }
    dropZone.addEventListener('drop', e => {
      if (e.dataTransfer?.files?.length) addPendingFiles(e.dataTransfer.files);
    });

    let currentImportRequest = null;
    let importCanceled = false;

    async function doImport(review) {
      const selectedCats = selectedCategoryIds(modalRoot.querySelector('#importCats'));
      const progress = modalRoot.querySelector('#importProgress');
      const bar = progress.firstElementChild;
      const status = modalRoot.querySelector('#importStatus');
      const cancelBtn = modalRoot.querySelector('#importCancelBtn');
      progress.classList.remove('hidden');
      progress.classList.add('indeterminate');
      const importedIds = [];
      const errors = [];
      let jobs = [];
      if (activeTab === 'local') jobs = pendingLocalFiles.map(file => ({ kind:'file', file }));
      else jobs = modalRoot.querySelector('#urlInput').value.split(/\r?\n/).map(x => x.trim()).filter(Boolean).map(url => ({ kind:'url', url }));
      if (!jobs.length) { status.textContent = 'Choose at least one file or URL.'; status.classList.add('bad'); return; }
      importCanceled = false;
      modalRoot.querySelectorAll('button, input, textarea').forEach(x => x.disabled = true);
      cancelBtn.disabled = false;
      cancelBtn.textContent = 'Cancel Import';
      cancelBtn.addEventListener('click', () => {
        importCanceled = true;
        if (currentImportRequest && typeof currentImportRequest.abort === 'function') currentImportRequest.abort();
        status.textContent = 'Canceling import…';
      }, { once: true });
      for (let i = 0; i < jobs.length; i++) {
        if (importCanceled) break;
        const job = jobs[i];
        status.textContent = `Importing ${i + 1} of ${jobs.length}… ${job.kind === 'file' ? job.file.name : job.url}`;
        bar.style.width = `${Math.round((i / jobs.length) * 100)}%`;
        try {
          let result;
          if (job.kind === 'file') {
            const form = new FormData();
            form.append('character_id', state.characterId);
            form.append('categories', JSON.stringify(selectedCats));
            form.append('file', job.file, job.file.name);
            result = await new Promise((resolve, reject) => {
              currentImportRequest = GM_xmlhttpRequest({
                method:'POST', url:`${API}/api/import/file`, data:form, timeout:300000, headers:{ Origin:'https://novelai.net' },
                onload:(res)=>{ currentImportRequest = null; if (res.status >= 200 && res.status < 300) { try { resolve(JSON.parse(res.responseText || '{}')); } catch (_) { resolve(null); } } else { try { reject(new Error(JSON.parse(res.responseText || '{}').error || `Request failed (${res.status})`)); } catch (_) { reject(new Error(`Request failed (${res.status})`)); } } },
                onerror:()=>{ currentImportRequest = null; reject(new Error(importCanceled ? 'Import canceled' : 'Could not connect to the local NovelAI Media Library program.')); },
                ontimeout:()=>{ currentImportRequest = null; reject(new Error(importCanceled ? 'Import canceled' : 'The local media request timed out.')); },
              });
            });
          } else {
            result = await gmRequest({ method:'POST', path:'/api/import/url', headers:{'Content-Type':'application/json'}, data:JSON.stringify({ character_id: state.characterId, categories:selectedCats, url:job.url }), timeout:300000 });
          }
          if (result?.item?.id && !importedIds.includes(result.item.id)) importedIds.push(result.item.id);
        } catch (err) {
          if (importCanceled && /canceled/i.test(String(err.message || ''))) break;
          errors.push(`${job.kind === 'file' ? job.file.name : job.url}: ${err.message}`);
        }
      }
      currentImportRequest = null;
      progress.classList.remove('indeterminate');
      bar.style.width = '100%';
      await loadLibrary(false);
      if (importCanceled) {
        status.textContent = `Import canceled. Imported ${importedIds.length} item(s).`;
        modalRoot.querySelectorAll('button, input, textarea').forEach(x => x.disabled = false);
        cancelBtn.textContent = 'Close';
        cancelBtn.addEventListener('click', close, { once: true });
        return;
      }
      status.textContent = `Imported ${importedIds.length} item(s)${errors.length ? `; ${errors.length} failed` : ''}.`;
      if (errors.length) console.warn('NovelAI Media import errors:', errors);
      if (review && importedIds.length) { close(); startReview(importedIds); }
      else { setTimeout(close, 650); }
    }

    modalRoot.querySelector('#importDirect').addEventListener('click', () => doImport(false));
    modalRoot.querySelector('#importReview').addEventListener('click', () => doImport(true));
  }

  function startReview(ids) {
    state.reviewQueue = ids.filter(id => state.library.media.some(m => m.id === id));
    state.reviewIndex = 0;
    state.carryCategories = null;
    if (!state.reviewQueue.length) return;
    renderReviewModal();
  }

  async function renderReviewModal() {
    const id = state.reviewQueue[state.reviewIndex];
    const m = state.library.media.find(x => x.id === id);
    if (!m) return finishReview();
    const c = state.library.characters.find(x => x.id === m.character_id);
    let selected = state.carryReviewCategories && state.carryCategories ? [...state.carryCategories] : [...(m.categories || [])];
    modalRoot.innerHTML = `
      <div class="modalWrap">
        <div class="modal large">
          <div class="modalHead">
            <div class="modalTitle">Rapid review — ${escapeHtml(c?.name || '')}</div>
            <div class="muted">${state.reviewIndex + 1} / ${state.reviewQueue.length}</div>
            <button class="icon ghost" data-close>×</button>
          </div>
          <div class="reviewBody">
            <div class="reviewStage" id="reviewStage"><div class="muted">Loading…</div></div>
            <div class="reviewSide">
              <div style="font:700 13px system-ui; margin-bottom:4px">Categories</div>
              <div class="muted" style="margin-bottom:10px">${m.media_type === 'video' ? 'Videos only appear in the automatic Videos category.' : 'Press 1–9 to toggle the first nine categories. Enter or → saves and advances.'}</div>
              <div id="reviewCats"></div>
              ${m.media_type === 'video' ? '' : '<button id="reviewAddCat">+ Create Category</button>'}
              <label class="check" style="margin-top:14px"><input type="checkbox" id="carryCats" ${state.carryReviewCategories ? 'checked' : ''}> Carry this category selection to the next image</label>
              <div class="muted" style="margin-top:12px; word-break:break-word">${escapeHtml(m.original_name || '')}</div>
            </div>
          </div>
          <div class="modalFoot">
            <button id="reviewDelete" class="danger">Delete media</button>
            <div class="grow"></div>
            <button id="reviewPrev">← Save + Previous</button>
            <button class="primary" id="reviewNext">Save + Next →</button>
          </div>
        </div>
      </div>`;

    const wrap = modalRoot.querySelector('.modalWrap');
    const rs = modalRoot.querySelector('#reviewStage');
    const reviewCats = modalRoot.querySelector('#reviewCats');

    function paintCats() {
      reviewCats.innerHTML = '';
      if (m.media_type === 'video') return;
      (c?.categories || []).forEach((cat, idx) => {
        const label = document.createElement('label');
        label.className = 'reviewCat';
        label.innerHTML = `<input type="checkbox" data-catid="${cat.id}" ${selected.includes(cat.id) ? 'checked' : ''}> <span>${escapeHtml(cat.name)}</span>${idx < 9 ? `<span class="key">${idx+1}</span>` : ''}`;
        reviewCats.appendChild(label);
      });
    }
    paintCats();

    let reviewUrl = null;
    try {
      if (m.media_type === 'video') {
        const v = document.createElement('video');
        v.src = directOriginalUrl(m.id); v.controls = true; v.loop = true; v.volume = 0.08; rs.innerHTML = ''; rs.appendChild(v);
      } else {
        const blob = await gmRequest({ path:`/api/media/${m.id}/original`, responseType:'blob', timeout:300000 });
        reviewUrl = URL.createObjectURL(blob);
        const img = document.createElement('img'); img.src = reviewUrl; rs.innerHTML = ''; rs.appendChild(img);
      }
    } catch (err) { rs.innerHTML = `<div class="muted">${escapeHtml(err.message)}</div>`; }

    const cleanup = () => { if (reviewUrl) URL.revokeObjectURL(reviewUrl); };
    modalRoot.querySelector('[data-close]').addEventListener('click', () => { cleanup(); finishReview(); });
    wrap.addEventListener('click', e => { if (e.target === wrap) { cleanup(); finishReview(); } });

    modalRoot.querySelector('#carryCats').addEventListener('change', (e) => { state.carryReviewCategories = e.target.checked; });
    const reviewAddCatBtn = modalRoot.querySelector('#reviewAddCat');
    if (reviewAddCatBtn) reviewAddCatBtn.addEventListener('click', async () => {
        selected = [...reviewCats.querySelectorAll('input[data-catid]:checked')].map(x => x.dataset.catid);
        const cat = await createCategoryFlow();
        if (!cat) return;
        c.categories = currentCharacter()?.categories || c.categories;
        paintCats();
      });

    async function saveCurrent() {
      if (m.media_type === 'video') {
        state.carryCategories = [];
        return;
      }
      selected = [...reviewCats.querySelectorAll('input[data-catid]:checked')].map(x => x.dataset.catid);
      const updated = await gmRequest({ method:'POST', path:`/api/media/${m.id}/categories`, headers:{'Content-Type':'application/json'}, data:JSON.stringify({ categories:selected }) });
      Object.assign(m, updated);
      state.carryCategories = [...selected];
    }

    async function move(delta) {
      try {
        await saveCurrent();
        cleanup();
        state.reviewIndex += delta;
        if (state.reviewIndex < 0) state.reviewIndex = 0;
        if (state.reviewIndex >= state.reviewQueue.length) return finishReview();
        renderReviewModal();
      } catch (err) { alert(err.message); }
    }

    modalRoot.querySelector('#reviewNext').addEventListener('click', () => move(1));
    modalRoot.querySelector('#reviewPrev').addEventListener('click', () => move(-1));
    modalRoot.querySelector('#reviewDelete').addEventListener('click', async () => {
      if (!confirm('Delete this media item from the managed library?')) return;
      try {
        await gmRequest({ method:'DELETE', path:`/api/media/${m.id}` });
        cleanup();
        state.reviewQueue.splice(state.reviewIndex, 1);
        await loadLibrary(false);
        if (!state.reviewQueue.length || state.reviewIndex >= state.reviewQueue.length) return finishReview();
        renderReviewModal();
      } catch (err) { alert(err.message); }
    });

    const keyHandler = (e) => {
      if (!modalRoot.childElementCount) return document.removeEventListener('keydown', keyHandler, true);
      if (e.target.matches('input, textarea, select')) return;
      if (e.key >= '1' && e.key <= '9' && m.media_type !== 'video') {
        const i = Number(e.key) - 1;
        const boxes = [...reviewCats.querySelectorAll('input[data-catid]')];
        if (boxes[i]) { e.preventDefault(); boxes[i].checked = !boxes[i].checked; }
      } else if (e.key === 'Enter' || e.key === 'ArrowRight') { e.preventDefault(); move(1); }
      else if (e.key === 'ArrowLeft') { e.preventDefault(); move(-1); }
    };
    document.addEventListener('keydown', keyHandler, true);
  }

  function finishReview() {
    modalRoot.innerHTML = '';
    state.reviewQueue = [];
    state.reviewIndex = 0;
    loadLibrary();
  }

  loadLibrary(false);
})();
