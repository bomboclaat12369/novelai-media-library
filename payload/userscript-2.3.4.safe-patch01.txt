
// NovelAI Media Library v2.3.4 safe restoration patch
// Restores thumbnail capture/display, timestamp playback-state preservation,
// split-view sizing, narrower right panel, and split fullscreen.
(() => {
  'use strict';

  const API = 'http://127.0.0.1:8765';
  const host = document.getElementById('nai-media-host');
  const root = host?.shadowRoot;
  if (!root || document.documentElement.dataset.naiMediaV234Safe === '1') return;
  document.documentElement.dataset.naiMediaV234Safe = '1';

  // IMPORTANT: intentionally no MutationObserver here. A previous patch repeatedly
  // mutated nodes from inside a subtree observer, which could create a render loop.
  const style = document.createElement('style');
  style.id = 'nai-media-v234-safe-style';
  style.textContent = `
    :host { --nai-right-edge: 74vw !important; }

    /* In split mode, let the two media slots use essentially the full viewer height. */
    .viewer { position:relative !important; }
    .stages.split { position:absolute !important; inset:0 !important; gap:2px !important; z-index:1 !important; }
    .stages.split .stage { flex:1 1 50% !important; min-height:0 !important; }
    .stages.split ~ .viewerNav {
      position:absolute !important; left:50% !important; bottom:5px !important;
      transform:translateX(-50%) !important; z-index:5 !important;
      padding:4px 7px !important; border-radius:10px !important;
      background:rgba(11,13,29,.80) !important; backdrop-filter:blur(4px) !important;
    }
    .stages.split ~ .badgeRow,
    .stages.split ~ .viewerMeta,
    .stages.split ~ .videoTools.visible { display:none !important; }

    /* When a video has a saved thumbnail, expose the image instead of covering it. */
    .tile:has(> img[data-mid]) .videoMark {
      inset:auto 6px 6px 6px !important;
      width:max-content !important; max-width:calc(100% - 12px) !important;
      height:auto !important; min-height:0 !important;
      display:block !important; padding:2px 6px !important;
      border-radius:999px !important;
      background:rgba(0,0,0,.68) !important;
      pointer-events:none !important;
      overflow:hidden !important;
      text-overflow:ellipsis !important;
      white-space:nowrap !important;
      font:700 10px system-ui !important;
    }
    .tile:has(> img[data-mid]) .videoMark .videoName { display:none !important; }

    .naiSplitLightbox {
      position:fixed !important; inset:0 !important; z-index:2147483950 !important;
      background:rgba(0,0,0,.96) !important;
      display:grid !important; grid-template-columns:repeat(2,minmax(0,1fr)) !important;
      gap:8px !important; padding:18px 24px !important;
    }
    .naiSplitLightbox.single { grid-template-columns:minmax(0,1fr) !important; }
    .naiSplitLightboxSlot {
      min-width:0 !important; min-height:0 !important;
      display:flex !important; align-items:center !important; justify-content:center !important;
      overflow:hidden !important; background:#05060d !important;
    }
    .naiSplitLightboxSlot img,
    .naiSplitLightboxSlot video {
      width:100% !important; height:100% !important;
      max-width:100% !important; max-height:100% !important;
      object-fit:contain !important; display:block !important;
    }
    .naiSplitLightboxClose {
      position:fixed !important; top:12px !important; right:14px !important;
      z-index:2147483960 !important; width:40px !important; height:40px !important;
      padding:0 !important; display:grid !important; place-items:center !important;
      border-radius:50% !important; background:rgba(22,24,47,.88) !important;
      font-size:24px !important;
    }
  `;
  root.appendChild(style);

  function gmRequest({ method = 'GET', path, data = null, headers = {}, responseType = 'text', timeout = 300000 }) {
    return new Promise((resolve, reject) => {
      GM_xmlhttpRequest({
        method,
        url: `${API}${path}`,
        data,
        headers: { Origin: 'https://novelai.net', ...headers },
        responseType,
        timeout,
        onload: res => {
          if (res.status >= 200 && res.status < 300) {
            if (responseType === 'blob') return resolve(res.response);
            const text = res.responseText || '';
            if (!text) return resolve(null);
            try { resolve(JSON.parse(text)); } catch (_) { resolve(text); }
            return;
          }
          let message = `Request failed (${res.status})`;
          try { message = JSON.parse(res.responseText || '{}').error || message; } catch (_) {}
          reject(new Error(message));
        },
        onerror: () => reject(new Error('Could not connect to the local NovelAI Media Library program.')),
        ontimeout: () => reject(new Error('The local media request timed out.')),
      });
    });
  }

  function waitFor(video, eventName, timeout = 30000) {
    return new Promise((resolve, reject) => {
      let timer = 0;
      const cleanup = () => {
        clearTimeout(timer);
        video.removeEventListener(eventName, done);
        video.removeEventListener('error', fail);
      };
      const done = () => { cleanup(); resolve(); };
      const fail = () => { cleanup(); reject(new Error('Could not load the video frame for thumbnail capture.')); };
      timer = setTimeout(() => { cleanup(); reject(new Error(`Timed out waiting for video ${eventName}.`)); }, timeout);
      video.addEventListener(eventName, done, { once:true });
      video.addEventListener('error', fail, { once:true });
    });
  }

  async function captureFrameFromSafeBlob(mediaId, seconds) {
    const sourceBlob = await gmRequest({ path:`/api/media/${mediaId}/original`, responseType:'blob' });
    const objectUrl = URL.createObjectURL(sourceBlob);
    const video = document.createElement('video');
    video.preload = 'auto';
    video.muted = true;
    video.playsInline = true;
    video.src = objectUrl;

    try {
      if (video.readyState < 1) await waitFor(video, 'loadedmetadata');
      const duration = Number.isFinite(video.duration) ? video.duration : 0;
      const target = duration > 0
        ? Math.min(Math.max(0, seconds), Math.max(0, duration - 0.01))
        : Math.max(0, seconds);

      if (Math.abs(video.currentTime - target) > 0.001) {
        video.currentTime = target;
        await waitFor(video, 'seeked');
      } else if (video.readyState < 2) {
        await waitFor(video, 'loadeddata');
      }

      if (!video.videoWidth || !video.videoHeight) throw new Error('Video frame is not ready yet.');
      const maxSide = 960;
      const scale = Math.min(1, maxSide / Math.max(video.videoWidth, video.videoHeight));
      const canvas = document.createElement('canvas');
      canvas.width = Math.max(1, Math.round(video.videoWidth * scale));
      canvas.height = Math.max(1, Math.round(video.videoHeight * scale));
      const ctx = canvas.getContext('2d');
      if (!ctx) throw new Error('Could not capture this video frame.');
      ctx.drawImage(video, 0, 0, canvas.width, canvas.height);

      return await new Promise((resolve, reject) => {
        canvas.toBlob(blob => blob ? resolve(blob) : reject(new Error('Could not create thumbnail image.')), 'image/jpeg', 0.9);
      });
    } finally {
      video.pause();
      video.removeAttribute('src');
      video.load();
      URL.revokeObjectURL(objectUrl);
    }
  }

  function activeVideo() {
    return root.querySelector('.stage.slot-active video') || root.querySelector('.stage:not(.hidden) video');
  }

  function activeMediaId(video = activeVideo()) {
    if (video) {
      const src = String(video.currentSrc || video.src || '');
      const match = src.match(/\/api\/media\/([0-9a-f]+)\/original(?:$|[?#])/i);
      if (match?.[1]) return match[1];
    }
    return root.querySelector('.tile.active')?.dataset?.id || null;
  }

  const thumbObjectUrls = new Map();
  async function showSavedThumbnail(mediaId) {
    if (!mediaId) return;
    const tile = root.querySelector(`.tile[data-id="${mediaId}"]`);
    if (!tile) return;
    try {
      const blob = await gmRequest({ path:`/api/media/${mediaId}/thumb`, responseType:'blob' });
      if (!blob || !blob.size || !String(blob.type || '').startsWith('image/')) return;
      const old = thumbObjectUrls.get(mediaId);
      if (old) URL.revokeObjectURL(old);
      const url = URL.createObjectURL(blob);
      thumbObjectUrls.set(mediaId, url);

      let img = tile.querySelector('img[data-mid]');
      if (!img) {
        img = document.createElement('img');
        img.loading = 'lazy';
        img.dataset.mid = mediaId;
        tile.insertBefore(img, tile.firstChild);
      }
      img.src = url;
    } catch (_) {}
  }

  // Override only the broken thumbnail button; do not continuously watch/mutate the grid.
  root.addEventListener('click', async event => {
    const button = event.target.closest?.('#setThumbBtn');
    if (!button) return;

    event.preventDefault();
    event.stopImmediatePropagation();

    const video = activeVideo();
    const mediaId = activeMediaId(video);
    if (!video || !mediaId) return;

    button.disabled = true;
    button.textContent = 'Saving thumbnail…';
    try {
      const seconds = Number.isFinite(video.currentTime) ? video.currentTime : 0;
      const frameBlob = await captureFrameFromSafeBlob(mediaId, seconds);
      await gmRequest({
        method:'POST',
        path:`/api/media/${mediaId}/thumbnail?seconds=${encodeURIComponent(seconds)}`,
        headers:{ 'Content-Type':'image/jpeg' },
        data:frameBlob,
      });
      await showSavedThumbnail(mediaId);
      button.textContent = 'Thumbnail saved';
    } catch (err) {
      alert(err?.message || String(err));
      button.textContent = 'Set thumbnail to current frame';
    } finally {
      setTimeout(() => {
        if (!button.isConnected) return;
        button.disabled = false;
        if (button.textContent === 'Thumbnail saved') button.textContent = 'Set thumbnail to current frame';
      }, 700);
    }
  }, true);

  // Preserve the video's play/pause state when jumping to T1/T2/T3.
  root.addEventListener('click', event => {
    const markerButton = event.target.closest?.('[data-seek-marker]');
    if (!markerButton) return;
    const video = activeVideo();
    if (!video) return;
    const wasPaused = video.paused;
    setTimeout(() => {
      if (!video.isConnected) return;
      if (wasPaused) video.pause();
      else video.play().catch(() => {});
    }, 40);
  }, true);

  function closeSplitLightbox() {
    const lightRoot = root.getElementById('lightRoot');
    if (!lightRoot) return;
    lightRoot.querySelectorAll('video[data-nai-split-full]').forEach(fullVideo => {
      const original = fullVideo._naiOriginalVideo;
      if (original?.isConnected) {
        try { original.currentTime = fullVideo.currentTime; } catch (_) {}
        if (!fullVideo.paused) original.play().catch(() => {});
      }
      try { fullVideo.pause(); } catch (_) {}
    });
    lightRoot.innerHTML = '';
  }

  function makeFullscreenCopy(source) {
    if (source.tagName === 'IMG') {
      const img = document.createElement('img');
      img.src = source.currentSrc || source.src;
      img.alt = source.alt || '';
      return img;
    }

    if (source.tagName === 'VIDEO') {
      const video = document.createElement('video');
      const wasPlaying = !source.paused;
      const startTime = Number.isFinite(source.currentTime) ? source.currentTime : 0;
      video.dataset.naiSplitFull = '1';
      video.controls = true;
      video.loop = true;
      video.playsInline = true;
      video.volume = Number.isFinite(source.volume) ? source.volume : 0.08;
      video.muted = source.muted;
      video._naiOriginalVideo = source;
      video.src = source.currentSrc || source.src;
      source.pause();
      video.addEventListener('loadedmetadata', () => {
        try {
          const end = Number.isFinite(video.duration) && video.duration > 0 ? Math.max(0, video.duration - 0.05) : startTime;
          video.currentTime = Math.min(startTime, end);
        } catch (_) {}
        if (wasPlaying) video.play().catch(() => {});
      }, { once:true });
      return video;
    }

    return null;
  }

  function openSplitLightbox() {
    const stages = root.getElementById('stages');
    if (!stages?.classList.contains('split')) return false;

    const sources = [root.getElementById('stage1'), root.getElementById('stage2')]
      .map(stage => stage?.querySelector('img,video'))
      .filter(Boolean)
      .slice(0, 2);
    if (!sources.length) return false;

    const lightRoot = root.getElementById('lightRoot');
    if (!lightRoot) return false;
    lightRoot.innerHTML = '';

    const box = document.createElement('div');
    box.className = `naiSplitLightbox${sources.length === 1 ? ' single' : ''}`;

    for (const source of sources) {
      const pane = document.createElement('div');
      pane.className = 'naiSplitLightboxSlot';
      const copy = makeFullscreenCopy(source);
      if (copy) pane.appendChild(copy);
      box.appendChild(pane);
    }

    const close = document.createElement('button');
    close.type = 'button';
    close.className = 'naiSplitLightboxClose';
    close.textContent = '×';
    close.title = 'Close';
    close.addEventListener('click', e => {
      e.preventDefault();
      e.stopPropagation();
      closeSplitLightbox();
    });
    box.appendChild(close);
    box.addEventListener('click', e => { if (e.target === box) closeSplitLightbox(); });
    lightRoot.appendChild(box);
    return true;
  }

  // In split mode, both the fullscreen button and clicking either image open both panes.
  root.addEventListener('click', event => {
    const stages = root.getElementById('stages');
    if (!stages?.classList.contains('split')) return;
    const fullButton = event.target.closest?.('#fullBtn');
    const stageImage = event.target.closest?.('.stage img');
    if (!fullButton && !stageImage) return;

    event.preventDefault();
    event.stopImmediatePropagation();
    openSplitLightbox();
  }, true);

  document.addEventListener('keydown', event => {
    if (event.key === 'Escape' && root.querySelector('.naiSplitLightbox')) closeSplitLightbox();
  }, true);

  window.addEventListener('beforeunload', () => {
    for (const url of thumbObjectUrls.values()) URL.revokeObjectURL(url);
    thumbObjectUrls.clear();
  }, { once:true });
})();
