// NovelAI Media Library v2.8.8: bridge temporary review drafts to the canonical crop editor
(() => {
  'use strict';

  const API = 'http://127.0.0.1:8765';
  const UI_STATE_KEY = 'nai-media-v1-ui';
  const host = document.getElementById('nai-media-host');
  const root = host?.shadowRoot;
  if (!root || document.documentElement.dataset.naiMediaV254 === '1') return;
  document.documentElement.dataset.naiMediaV254 = '1';

  const modalRoot = root.getElementById('modalRoot');
  const cropBtn = root.getElementById('naiCropBtn');
  const stageEls = [root.getElementById('stage1'), root.getElementById('stage2')];
  const charSelect = root.getElementById('characterSelect');
  if (!modalRoot || !cropBtn || !stageEls[0] || !charSelect) return;

  const previewUrls = new Set();
  let bridge = null;
  let decorateQueued = false;

  const style = document.createElement('style');
  style.id = 'nai-media-v254-style';
  style.textContent = `
    #naiReviewCropBtn { display:none !important; }
    #naiReviewCropBtnV254 { margin-left:6px !important; white-space:nowrap !important; }
    #naiReviewCropBtnV254.hasCrop { background:#4c467f !important; border-color:#8179d8 !important; }
  `;
  root.appendChild(style);

  function uiState() {
    try { return JSON.parse(localStorage.getItem(UI_STATE_KEY) || '{}') || {}; }
    catch (_) { return {}; }
  }

  function saveUiState(value) {
    try { localStorage.setItem(UI_STATE_KEY, JSON.stringify(value || {})); } catch (_) {}
  }

  function gmRequest({method='GET', path, data=null, headers={}, responseType='text', timeout=120000}) {
    return new Promise((resolve, reject) => GM_xmlhttpRequest({
      method,
      url:`${API}${path}`,
      data,
      headers:{Origin:'https://novelai.net', ...headers},
      responseType,
      timeout,
      onload:res => {
        if (res.status >= 200 && res.status < 300) {
          if (responseType === 'blob') return resolve(res.response);
          const text = res.responseText || '';
          if (!text) return resolve(null);
          try { return resolve(JSON.parse(text)); } catch (_) { return resolve(text); }
        }
        let message = `Request failed (${res.status})`;
        try { message = JSON.parse(res.responseText || '{}').error || message; } catch (_) {}
        reject(new Error(message));
      },
      onerror:()=>reject(new Error('Could not connect to the local NovelAI Media Library program.')),
      ontimeout:()=>reject(new Error('The local media request timed out.')),
    }));
  }

  async function sha256Blob(blob) {
    const digest = await crypto.subtle.digest('SHA-256', await blob.arrayBuffer());
    return [...new Uint8Array(digest)].map(x => x.toString(16).padStart(2, '0')).join('');
  }

  async function resolveReviewMedia(stage) {
    const existing = stage.dataset.naiReviewMediaId;
    const draft = root.naiReviewDrafts?.get(existing);
    if (draft) return draft.media;
    const lib = await gmRequest({path:'/api/library'});
    if (existing) {
      const item = (lib?.media || []).find(m => m.id === existing) || null;
      if (item) return item;
    }

    const img = stage.querySelector('img');
    if (!img) return null;
    const src = img.currentSrc || img.src;
    if (!src) return null;
    const response = await fetch(src);
    if (!response.ok) throw new Error('Could not read the Rapid Review image.');
    const sha = await sha256Blob(await response.blob());
    const characterId = charSelect.value || '';
    const item = (lib?.media || []).find(m =>
      m?.media_type === 'image' && m?.sha256 === sha && (!characterId || m.character_id === characterId)
    ) || null;
    if (item?.id) stage.dataset.naiReviewMediaId = item.id;
    return item;
  }

  function normalizeCrop(media) {
    let top = Number(media?.crop_top || 0);
    let bottom = Number(media?.crop_bottom || 0);
    if (!Number.isFinite(top)) top = 0;
    if (!Number.isFinite(bottom)) bottom = 0;
    top = Math.max(0, Math.min(.89, top));
    bottom = Math.max(0, Math.min(.89, bottom));
    if (top + bottom > .90) {
      const scale = .90 / (top + bottom);
      top *= scale;
      bottom *= scale;
    }
    return {top, bottom};
  }

  function setButtonState(button, media) {
    if (!button || !media) return;
    const crop = normalizeCrop(media);
    const has = crop.top > 0 || crop.bottom > 0;
    button.classList.toggle('hasCrop', has);
    button.textContent = has ? 'Crop image ✓' : 'Crop image';
    button.title = has ? 'Adjust the saved crop for this image' : 'Crop the top or bottom of this image';
  }

  async function makePreviewBlob(originalBlob, crop) {
    if (!crop.top && !crop.bottom) return originalBlob;
    const bitmap = await createImageBitmap(originalBlob);
    try {
      const y1 = Math.max(0, Math.min(bitmap.height - 1, Math.round(bitmap.height * crop.top)));
      const y2 = Math.max(y1 + 1, Math.min(bitmap.height, Math.round(bitmap.height * (1 - crop.bottom))));
      const canvas = document.createElement('canvas');
      canvas.width = bitmap.width;
      canvas.height = y2 - y1;
      const ctx = canvas.getContext('2d');
      if (!ctx) throw new Error('Could not create crop preview.');
      ctx.drawImage(bitmap, 0, y1, bitmap.width, y2 - y1, 0, 0, bitmap.width, y2 - y1);
      const type = originalBlob.type === 'image/png' ? 'image/png' : (originalBlob.type === 'image/webp' ? 'image/webp' : 'image/jpeg');
      return await new Promise((resolve, reject) => canvas.toBlob(
        b => b ? resolve(b) : reject(new Error('Could not create crop preview.')),
        type,
        type === 'image/png' ? undefined : .96
      ));
    } finally {
      bitmap.close();
    }
  }

  async function refreshRapidPreview(stage, mediaId, button) {
    if (!stage?.isConnected) return;
    try {
      const draft = root.naiReviewDrafts?.get(mediaId);
      const lib = draft ? null : await gmRequest({path:'/api/library'});
      const media = draft?.media || (lib?.media || []).find(m => m.id === mediaId) || null;
      if (!media) return;
      if (button?.isConnected) setButtonState(button, media);
      const img = stage.querySelector('img');
      if (!img) return;
      const original = draft ? draft.file : await gmRequest({
        path:`/api/media/${encodeURIComponent(mediaId)}/original`,
        responseType:'blob',
        timeout:300000,
      });
      const preview = await makePreviewBlob(original, normalizeCrop(media));
      if (!stage.isConnected) return;
      const url = URL.createObjectURL(preview);
      previewUrls.add(url);
      const old = stage._naiV254PreviewUrl;
      if (old) {
        URL.revokeObjectURL(old);
        previewUrls.delete(old);
      }
      stage._naiV254PreviewUrl = url;
      if (img.isConnected) img.src = url;
      stage.dataset.naiReviewMediaId = mediaId;
    } catch (err) {
      console.warn('[NovelAI Media] Could not refresh Rapid Review crop preview', err);
    }
  }

  root.naiRefreshReviewCrop = refreshRapidPreview;
  root.naiReleaseReviewCrop = stage => {
    const url = stage?._naiV254PreviewUrl;
    if (!url) return;
    URL.revokeObjectURL(url);
    previewUrls.delete(url);
    stage._naiV254PreviewUrl = null;
  };

  function restoreBridge() {
    if (!bridge) return;
    const current = bridge;
    bridge = null;
    try { current.observer?.disconnect(); } catch (_) {}
    clearTimeout(current.timeout);
    saveUiState(current.savedState);
    current.proxy.remove();
    if (current.button?.isConnected) current.button.disabled = false;
    refreshRapidPreview(current.reviewStage, current.mediaId, current.button);
  }

  function watchCanonicalEditor(current) {
    let sawEditor = false;
    current.observer = new MutationObserver(() => {
      const open = !!root.querySelector('.naiCropWrap');
      if (open) {
        sawEditor = true;
        return;
      }
      if (sawEditor) restoreBridge();
    });
    current.observer.observe(root, {childList:true});
    current.timeout = setTimeout(() => {
      if (root.querySelector('.naiCropWrap')) return;
      const button = current.button;
      restoreBridge();
      if (button?.isConnected) alert('The Crop image screen did not open. Please try again.');
    }, 15000);
  }

  async function openCanonicalCrop(stage, button) {
    if (bridge || button.disabled) return;
    button.disabled = true;
    const oldText = button.textContent;
    button.textContent = 'Opening crop…';

    try {
      const media = await resolveReviewMedia(stage);
      if (!media?.id) throw new Error('Could not identify this Rapid Review image.');
      if (media.media_type !== 'image') return;

      const savedState = uiState();
      const slot = stageEls[1]?.classList.contains('slot-active') ? 1 : 0;
      const tempState = {...savedState};
      const ids = Array.isArray(savedState.slotMediaIds) ? [...savedState.slotMediaIds] : [null, null];
      while (ids.length < 2) ids.push(null);
      ids[slot] = media.id;
      tempState.activeSlot = slot;
      tempState.selectedMediaId = media.id;
      tempState.slotMediaIds = ids;
      saveUiState(tempState);

      // The canonical crop editor only needs an image element in the active stage.
      // Put a hidden proxy first so its internal crop application never touches the
      // real viewer image/video behind Rapid Review.
      const proxy = document.createElement('img');
      proxy.alt = '';
      proxy.setAttribute('aria-hidden', 'true');
      proxy.style.cssText = 'display:none!important;width:0!important;height:0!important;position:absolute!important;';
      stageEls[slot].insertBefore(proxy, stageEls[slot].firstChild);

      bridge = {
        savedState,
        slot,
        proxy,
        mediaId:media.id,
        reviewStage:stage,
        button,
        observer:null,
        timeout:0,
      };

      // Allow the v2.4.1 stage observer to see the proxy and refresh the Crop button,
      // then invoke the exact same crop editor used by the normal image viewer.
      await new Promise(resolve => requestAnimationFrame(() => requestAnimationFrame(resolve)));
      cropBtn.disabled = false;
      watchCanonicalEditor(bridge);
      cropBtn.click();
    } catch (err) {
      if (bridge) restoreBridge();
      if (button.isConnected) {
        button.disabled = false;
        button.textContent = oldText || 'Crop image';
      }
      alert(err?.message || String(err));
    }
  }

  function decorateReview() {
    const stage = modalRoot.querySelector('#reviewStage');
    const side = modalRoot.querySelector('.reviewSide');
    if (!stage || !side) return;

    const oldButton = side.querySelector('#naiReviewCropBtn');
    if (oldButton) oldButton.style.display = 'none';

    let button = side.querySelector('#naiReviewCropBtnV254');
    if (stage.querySelector('video')) {
      button?.remove();
      return;
    }

    if (!button) {
      button = document.createElement('button');
      button.type = 'button';
      button.id = 'naiReviewCropBtnV254';
      button.textContent = 'Crop image';
      if (oldButton) oldButton.insertAdjacentElement('afterend', button);
      else {
        const addCategory = side.querySelector('#reviewAddCat');
        if (addCategory) addCategory.insertAdjacentElement('afterend', button);
        else side.insertBefore(button, side.firstChild);
      }
    }

    const img = stage.querySelector('img');
    button.disabled = !img || !!bridge;
    if (!img) return;

    const src = img.currentSrc || img.src || '';
    if (stage.dataset.naiV254ResolvedSrc === src && stage.dataset.naiReviewMediaId) return;
    stage.dataset.naiV254ResolvedSrc = src;
    resolveReviewMedia(stage).then(media => {
      if (media && button.isConnected && stage.isConnected) setButtonState(button, media);
    }).catch(() => {});
  }

  function scheduleDecorate() {
    if (decorateQueued) return;
    decorateQueued = true;
    requestAnimationFrame(() => {
      decorateQueued = false;
      decorateReview();
    });
  }

  // v2.5.2/v2.5.3 keep their original Crop button hidden so their older handlers stay
  // registered but never receive user clicks. This replacement uses the canonical editor.
  window.addEventListener('click', event => {
    const path = event.composedPath?.() || [];
    const button = path.find(node => node?.id === 'naiReviewCropBtnV254');
    if (!button || !path.includes(host)) return;
    event.preventDefault();
    event.stopImmediatePropagation();
    const stage = modalRoot.querySelector('#reviewStage');
    if (!stage || stage.querySelector('video')) return;
    openCanonicalCrop(stage, button);
  }, true);

  new MutationObserver(scheduleDecorate).observe(modalRoot, {childList:true, subtree:true});
  scheduleDecorate();

  window.addEventListener('pagehide', () => {
    for (const url of previewUrls) URL.revokeObjectURL(url);
    previewUrls.clear();
  }, {once:true});
})();
