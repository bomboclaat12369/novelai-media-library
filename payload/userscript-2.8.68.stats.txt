// Character totals and inexpensive source-resolution flags.
(() => {
  'use strict';
  const root=document.getElementById('nai-media-host')?.shadowRoot;
  if(!root)return;
  const select=root.getElementById('characterSelect'),modal=root.getElementById('modalRoot');
  const button=document.createElement('button');button.id='naiCharacterStats';button.textContent='Stats';button.title='File counts, storage, and resolution flags';button.style.cssText='font:11px system-ui;padding:5px 8px';
  select.insertAdjacentElement('afterend',button);
  const request=(path,body)=>new Promise((resolve,reject)=>GM_xmlhttpRequest({method:body===undefined?'GET':'POST',url:'http://127.0.0.1:8765'+path,timeout:30000,headers:body===undefined?{}:{'Content-Type':'application/json'},data:body===undefined?undefined:JSON.stringify(body),
    onload:r=>{try{const data=JSON.parse(r.responseText);if(r.status<200||r.status>=300)throw Error(data.error||'Restart the companion to enable statistics');resolve(data);}catch(e){reject(e);}},onerror:()=>reject(Error('Companion unavailable')),ontimeout:()=>reject(Error('Request timed out'))}));
  const format=bytes=>{if(bytes<1024)return bytes+' B';const i=Math.min(3,Math.floor(Math.log(bytes)/Math.log(1024)));return (bytes/1024**i).toFixed(2)+' '+['B','KiB','MiB','GiB'][i];};
  button.onclick=async()=>{
    const id=select.value,character=root.naiGetLibrary?.()?.characters?.find(c=>c.id===id);
    if(!character){alert('Choose a girl to see her statistics.');return;}
    if(modal.childElementCount)return;
    const wrap=document.createElement('div');wrap.className='modalWrap';
    wrap.innerHTML=`<div class="modal" style="width:min(620px,94vw);max-height:90vh;overflow:auto">
      <div class="modalHead"><div class="modalTitle"></div><button data-close>✕</button></div>
      <div style="padding:14px;display:grid;gap:14px">
        <div id="naiStatsCounts">Loading…</div>
        <div style="border-top:1px solid #ffffff22;padding-top:12px;display:grid;gap:9px">
          <b>Automatic resolution flags</b>
          <label><input type="checkbox" id="naiQualityEnabled"> Flag new images below the threshold as Needs replacement</label>
          <label>Threshold: <input type="number" id="naiQualityMP" min="0.01" max="100" step="0.01" style="width:90px;padding:5px 7px;border:1px solid #ffffff30;border-radius:5px;background:#101329;color:#fff"> megapixels</label>
          <div class="muted">Uses original width × height. Applies to all girls. Cleared flags stay cleared for that source image. No blur or watermark analysis.</div>
          <div style="display:flex;gap:8px;flex-wrap:wrap"><button id="naiQualitySave">Save setting</button><button id="naiQualityScan">Scan this girl's existing files</button></div>
          <div id="naiStatsMessage" class="muted" role="status"></div>
        </div>
      </div></div>`;
    wrap.querySelector('.modalTitle').textContent=character.name+' — Library statistics';modal.appendChild(wrap);
    const message=wrap.querySelector('#naiStatsMessage'),counts=wrap.querySelector('#naiStatsCounts'),enabled=wrap.querySelector('#naiQualityEnabled'),mp=wrap.querySelector('#naiQualityMP'),scan=wrap.querySelector('#naiQualityScan'),save=wrap.querySelector('#naiQualitySave');
    let timer=0,closed=false;const close=()=>{closed=true;clearTimeout(timer);wrap.remove();};
    wrap.querySelector('[data-close]').onclick=close;wrap.onclick=e=>{if(e.target===wrap)close();};
    function paint(s){
      counts.replaceChildren();
      const table=document.createElement('table');table.style.cssText='width:100%;border-collapse:collapse;text-align:right';
      const rows=[['','Images','Videos'],['Saved total',s.saved.images,s.saved.videos],['Inside Sets',s.in_sets.images,s.in_sets.videos],['Outside Sets',s.outside_sets.images,s.outside_sets.videos],['Awaiting review',s.pending.images,s.pending.videos]];
      for(const [i,row] of rows.entries()){const tr=document.createElement('tr');for(const [j,value]of row.entries()){const cell=document.createElement(i===0?'th':'td');cell.textContent=value;cell.style.cssText='padding:7px 6px;border-bottom:1px solid #ffffff16;'+(j===0?'text-align:left':'');tr.appendChild(cell);}table.appendChild(tr);}
      counts.appendChild(table);
      for(const text of [`${s.saved.images+s.saved.videos} unique saved files · ${s.sets} ${s.sets===1?'Set':'Sets'} · ${s.needs_replacement} ${s.needs_replacement===1?'image needs':'images need'} replacement`,
        `Source storage: ${format(s.image_bytes+s.video_bytes)} (${format(s.image_bytes)} images · ${format(s.video_bytes)} videos)`,
        'Storage includes pending files; excludes generated thumbnails, caches, and retained deleted files. Each saved file is counted once, even if it also appears in All.',
        ...(s.unknown_sizes?[`${s.unknown_sizes} files have unknown sizes. Scan existing files to complete storage totals.`]:[])]){const el=document.createElement('div');el.className='muted';el.style.marginTop='9px';el.textContent=text;counts.appendChild(el);}
    }
    const refresh=async()=>{const s=await request(`/api/characters/${id}/stats`);if(!closed)paint(s);};
    const setting=()=>{const value=Number(mp.value);if(!Number.isFinite(value)||value<.01||value>100)throw Error('Enter a threshold between 0.01 and 100 megapixels.');return {enabled:enabled.checked,min_pixels:Math.round(value*1000000)};};
    async function poll(){
      if(closed)return;
      try{const state=await request('/api/media/metadata-index');if(closed)return;
        if(state.running){message.textContent=`Scanning ${state.processed} / ${state.total} files…`;timer=setTimeout(poll,750);}
        else{await refresh();await root.naiReloadLibrary?.();message.textContent='Scan complete. Resolution flags and storage totals updated.';scan.disabled=save.disabled=false;}
      }catch(e){message.textContent=e.message;scan.disabled=save.disabled=false;}
    }
    save.onclick=async()=>{save.disabled=true;try{await request('/api/quality/settings',setting());message.textContent='Saved. New imports use this setting. Scan existing files to apply it to them.';}catch(e){message.textContent=e.message;}finally{save.disabled=false;}};
    scan.onclick=async()=>{scan.disabled=save.disabled=true;try{await request('/api/quality/settings',setting());await request('/api/quality/scan',{character_id:id});poll();}catch(e){message.textContent=e.message;scan.disabled=save.disabled=false;}};
    try{const [s,q,index]=await Promise.all([request(`/api/characters/${id}/stats`),request('/api/quality/settings'),request('/api/media/metadata-index')]);if(closed)return;paint(s);enabled.checked=q.enabled;mp.value=q.min_pixels/1000000;if(index.running){scan.disabled=save.disabled=true;poll();}}
    catch(e){counts.textContent=e.message;scan.disabled=save.disabled=true;}
  };
})();
