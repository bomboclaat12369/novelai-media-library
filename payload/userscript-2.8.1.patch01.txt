// NovelAI Media Library v2.8.1: non-invasive localhost/import diagnostics
(() => {
  'use strict';

  const API = 'http://127.0.0.1:8765';
  const host = document.getElementById('nai-media-host');
  const root = host?.shadowRoot;
  if (!root || document.documentElement.dataset.naiMediaV281 === '1') return;
  document.documentElement.dataset.naiMediaV281 = '1';

  const modalRoot = root.getElementById('modalRoot');
  if (!modalRoot) return;

  function gmJson(path, timeout=10000) {
    return new Promise((resolve, reject) => GM_xmlhttpRequest({
      method:'GET',
      url:`${API}${path}`,
      timeout,
      headers:{Origin:'https://novelai.net'},
      onload:r => {
        if (r.status >= 200 && r.status < 300) {
          try { resolve(JSON.parse(r.responseText || '{}')); }
          catch (_) { resolve(null); }
        } else reject(new Error(`HTTP ${r.status}`));
      },
      onerror:()=>reject(new Error('local request failed')),
      ontimeout:()=>reject(new Error('local request timed out')),
    }));
  }

  async function pingLocalhost() {
    const samples = [];
    for (let i=0;i<3;i++) {
      const t = performance.now();
      try {
        await gmJson('/api/health', 5000);
        samples.push(Math.round((performance.now()-t)*10)/10);
      } catch (_) {}
    }
    if (!samples.length) return;
    const sorted = [...samples].sort((a,b)=>a-b);
    const median = sorted[Math.floor(sorted.length/2)];
    console.info('[NovelAI Media diagnostics] startup', {
      loader: document.documentElement.dataset.naiMediaLoaderVersion || 'older/unknown',
      payload: localStorage.getItem('nai-media-github-payload-version') || 'unknown',
      localhost_health_ms: samples,
      localhost_health_median_ms: median,
    });
  }

  let lastStage = null;
  let queued = false;
  async function inspectAfterReviewOpen() {
    queued = false;
    const stage = modalRoot.querySelector('#reviewStage');
    if (!stage || stage === lastStage) return;
    lastStage = stage;
    await new Promise(r => setTimeout(r, 100));
    try {
      const last = await gmJson('/api/diagnostics/last-import', 5000);
      if (!last || last.state === 'none') return;
      const age = Date.now() - Number(last.recorded_at_epoch_ms || 0);
      if (!Number.isFinite(age) || age < 0 || age > 120000) return;
      console.info('[NovelAI Media diagnostics] last import', last);
      const total = Number(last?.debug_timing_ms?.server_total || 0);
      if (total > 2000) {
        try {
          const probe = await gmJson('/api/diagnostics/performance', 15000);
          console.info('[NovelAI Media diagnostics] local disk probe', probe);
        } catch (err) {
          console.warn('[NovelAI Media diagnostics] disk probe failed', err);
        }
      }
    } catch (_) {
      // Runtime older than the diagnostics build; remain completely passive.
    }
  }

  function scheduleInspect() {
    if (queued) return;
    queued = true;
    queueMicrotask(inspectAfterReviewOpen);
  }

  new MutationObserver(scheduleInspect).observe(modalRoot, {childList:true, subtree:true});
  setTimeout(pingLocalhost, 250);
})();
