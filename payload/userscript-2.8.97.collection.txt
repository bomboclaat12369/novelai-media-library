function ovr(entry,rows){
 const placed=rows.filter(e=>!e.provisional&&['Developing','Established'].includes(e.rating_status));
 if(entry.provisional||!entry.played||placed.length<5)return null;
 const score=Math.round(entry.rating*1e8),values=placed.map(e=>Math.round(e.rating*1e8)),lo=Math.min(...values),hi=Math.max(...values);if(hi===lo)return null;
 const mid=v=>values.filter(x=>x<v).length+(values.filter(x=>x===v).length-1)/2;
 const p=(mid(score)-mid(lo))/(mid(hi)-mid(lo));
 const anchors=[[0,60],[.5,78],[.8,87],[.9,91],[.97,95],[1,99]];
 let value=99;for(let i=1;i<anchors.length;i++)if(p<=anchors[i][0]){const [x,a]=anchors[i-1],[y,b]=anchors[i];value=a+(b-a)*(p-x)/(y-x);break;}
 return Math.min(placed.length<20?95:99,Math.max(60,Math.round(value)));
}
