// NovelAI Media Library v2.5.3: inline Rapid Review crop + clean viewer state on refresh
(() => {
  'use strict';

  const API = 'http://127.0.0.1:8765';
  const UI_STATE_KEY = 'nai-media-v1-ui';
  const host = document.getElementById('nai-media-host');
  const root = host?.shadowRoot;
  if (!root || document.documentElement.dataset.naiMediaV253 === '1') return;
  document.documentElement.dataset.naiMediaV253 = '1';

  const modalRoot = root.getElementById('modalRoot');
  const charSelect = root.getElementById('characterSelect');
  if (!modalRoot || !charSelect) return;

  // Page loads should always start with an empty single-image viewer. The base script
  // reads this object before this patch runs, so the first load after upgrading may
  // need one immediate reload if an old selected/split state was restored.
  function resetPersistedViewerState() {
    let state = {};
    try { state = JSON.parse(localStorage.getItem(UI_STATE_KEY) || '{}') || {}; } catch (_) {}
    const hadRestoredViewer = !!(
      state.splitView ||
      state.activeSlot === 1 ||
      state.selectedMediaId ||
      (Array.isArray(state.slotMediaIds) && state.slotMediaIds.some(Boolean))
    );
    state.splitView = false;
    state.activeSlot = 0;
    state.selectedMediaId = null;
    state.slotMediaIds = [null, null];
    try { localStorage.setItem(UI_STATE_KEY, JSON.stringify(state)); } catch (_) {}
    return hadRestoredViewer;
  }

  const hadRestoredViewer = resetPersistedViewerState();
  window.addEventListener('beforeunload', resetPersistedViewerState, {capture:true});
  window.addEventListener('pagehide', resetPersistedViewerState, {capture:true});

  if (hadRestoredViewer) {
    // A single controlled reload lets the already-running base script reinitialize from
    // the just-cleared state. The next load sees no selection, so this does not loop.
    setTimeout(() => location.reload(), 0);
    return;
  }

  const style = document.createElement('style');
  style.id = 'nai-media-v253-style';
  style.textContent = `
    .reviewStage.naiReviewCropActiveV253 { position:relative !important; }
    .naiReviewCropOverlayV253 {
      position:absolute; z-index:20; pointer-events:none; line-height:normal;
    }
    .naiReviewCropMaskV253 {
      position:absolute; left:0; right:0; background:rgba(0,0,0,.68); pointer-events:none;
    }
    .naiReviewCropMaskV253.top { top:0; }
    .naiReviewCropMaskV253.bottom { bottom:0; }
    .naiReviewCropHandleV253 {
      position:absolute; left:0; right:0; height:18px; transform:translateY(-9px);
      z-index:4; cursor:ns-resize; pointer-events:auto; touch-action:none;
    }
    .naiReviewCropHandleV253::before {
      content:''; position:absolute; left:0; right:0; top:8px; height:2px;
      background:#8c83ff; box-shadow:0 0 8px #8c83ff;
    }
    .naiReviewCropHandleV253::after {
      content:'↕'; position:absolute; left:50%; top:-4px; transform:translateX(-50%);
      width:30px; height:26px; border-radius:6px; background:#6155d9; color:#fff;
      display:grid; place-items:center; font:700 14px/1 system-ui;
    }
    .naiReviewCropControlsV253 {
      position:absolute; z-index:24; top:9px; right:9px;
      display:flex; align-items:center; gap:6px; padding:6px;
      border:1px solid rgba(255,255,255,.14); border-radius:8px;
      background:rgba(12,14,31,.9); box-shadow:0 2px 10px rgba(0,0,0,.35);
      pointer-events:auto;
    }
    .naiReviewCropControlsV253 button { padding:5px 8px !important; font-size:10px !important; }
    .naiReviewCropReadoutV253 {
      min-width:150px; color:#d7d9ea; text-align:center; font:600 10px/1.2 system-ui;
    }
    .naiReviewCropBusyV253 { opacity:.65; pointer-events:none; }
  `;
  root.appendChild(style);

  function normalizeCrop(value) {
    let top = Number(value?.top ?? value?.crop_top ?? 0);
    let bottom = Number(value?.bottom ?? value?.crop_bottom ?? 0);
    if (!Number.isFinite(top)) top = 0;
    if (!Number.isFinite(bottom)) bottom = 0;
    top = Math.max(0, Math.min(.89, top));
    bottom = Math.max(0, Math.min(.89, bottom));
    if (top + bottom > .90) {
      const scale = .90 / (top + bottom);
      top *= scale;
      bottom *= scale;
    }
    return {top, bottom};
  }

  function gmRequest({method='GET', path, data=null, headers={}, responseType='text', timeout=120000}) {
    return new Promise((resolve, reject) => GM_xmlhttpRequest({
      method,
      url:`${API}${path}`,
      data,
      headers:{Origin:'https://novelai.net', ...headers},
      responseType,
      timeout,
      onload:res => {
        if (res.status >= 200 && res.status < 300) {
          if (responseType === 'blob') return resolve(res.response);
          const text = res.responseText || '';
          if (!text) return resolve(null);
          try { return resolve(JSON.parse(text)); } catch (_) { return resolve(text); }
        }
        let message = `Request failed (${res.status})`;
        try { message = JSON.parse(res.responseText || '{}').error || message; } catch (_) {}
        reject(new Error(message));
      },
      onerror:()=>reject(new Error('Could not connect to the local NovelAI Media Library program.')),
      ontimeout:()=>reject(new Error('The local media request timed out.')),
    }));
  }

  async function sha256Blob(blob) {
    const digest = await crypto.subtle.digest('SHA-256', await blob.arrayBuffer());
    return [...new Uint8Array(digest)].map(x => x.toString(16).padStart(2, '0')).join('');
  }

  async function resolveReviewMedia(stage) {
    const lib = await gmRequest({path:'/api/library'});
    const existingId = stage.dataset.naiReviewMediaId;
    if (existingId) {
      const found = (lib?.media || []).find(m => m.id === existingId) || null;
      if (found) return found;
    }

    const img = stage.querySelector('img');
    if (!img) return null;
    const src = img.currentSrc || img.src;
    if (!src) return null;
    const response = await fetch(src);
    if (!response.ok) throw new Error('Could not read the Rapid Review image.');
    const sha = await sha256Blob(await response.blob());
    const characterId = charSelect.value || '';
    const media = (lib?.media || []).find(m =>
      m?.media_type === 'image' &&
      m?.sha256 === sha &&
      (!characterId || m.character_id === characterId)
    ) || null;
    if (media?.id) stage.dataset.naiReviewMediaId = media.id;
    return media;
  }

  function setReviewCropButtonState(button, media) {
    if (!button || !media) return;
    const crop = normalizeCrop(media);
    const cropped = crop.top > 0 || crop.bottom > 0;
    button.classList.toggle('hasCrop', cropped);
    button.textContent = cropped ? 'Crop image ✓' : 'Crop image';
    button.title = cropped ? 'Adjust the saved crop for this image' : 'Crop the top or bottom of this image';
  }

  let active = null;

  function paintActiveCrop() {
    if (!active) return;
    const topPct = active.top * 100;
    const bottomPct = active.bottom * 100;
    active.topMask.style.height = `${topPct}%`;
    active.bottomMask.style.height = `${bottomPct}%`;
    active.topHandle.style.top = `${topPct}%`;
    active.bottomHandle.style.top = `${100 - bottomPct}%`;
    active.readout.textContent = `Top ${topPct.toFixed(1)}% · Bottom ${bottomPct.toFixed(1)}%`;
  }

  function placeOverlay() {
    if (!active || !active.stage.isConnected || !active.img.isConnected) return;
    const sr = active.stage.getBoundingClientRect();
    const ir = active.img.getBoundingClientRect();
    active.overlay.style.left = `${Math.max(0, ir.left - sr.left)}px`;
    active.overlay.style.top = `${Math.max(0, ir.top - sr.top)}px`;
    active.overlay.style.width = `${Math.max(1, ir.width)}px`;
    active.overlay.style.height = `${Math.max(1, ir.height)}px`;
  }

  function dragCrop(edge, clientY) {
    if (!active) return;
    const rect = active.overlay.getBoundingClientRect();
    if (!rect.height) return;
    const p = Math.max(0, Math.min(1, (clientY - rect.top) / rect.height));
    if (edge === 'top') active.top = Math.max(0, Math.min(p, .90 - active.bottom));
    else active.bottom = Math.max(0, Math.min(1 - p, .90 - active.top));
    paintActiveCrop();
  }

  async function makeCroppedPreview(blob, crop) {
    const bitmap = await createImageBitmap(blob);
    try {
      const y1 = Math.max(0, Math.min(bitmap.height - 1, Math.round(bitmap.height * crop.top)));
      const y2 = Math.max(y1 + 1, Math.min(bitmap.height, Math.round(bitmap.height * (1 - crop.bottom))));
      const canvas = document.createElement('canvas');
      canvas.width = bitmap.width;
      canvas.height = y2 - y1;
      const ctx = canvas.getContext('2d');
      if (!ctx) throw new Error('Could not create crop preview.');
      ctx.drawImage(bitmap, 0, y1, bitmap.width, y2 - y1, 0, 0, bitmap.width, y2 - y1);
      const type = blob.type === 'image/png' ? 'image/png' : (blob.type === 'image/webp' ? 'image/webp' : 'image/jpeg');
      return await new Promise((resolve, reject) => canvas.toBlob(
        b => b ? resolve(b) : reject(new Error('Could not create crop preview.')),
        type,
        type === 'image/png' ? undefined : .96
      ));
    } finally {
      bitmap.close();
    }
  }

  function closeInlineCrop({restore=false}={}) {
    if (!active) return;
    const current = active;
    active = null;
    try { current.resizeObserver?.disconnect(); } catch (_) {}
    window.removeEventListener('pointermove', current.onMove, true);
    window.removeEventListener('pointerup', current.onUp, true);
    current.overlay.remove();
    current.controls.remove();
    current.stage.classList.remove('naiReviewCropActiveV253');
    if (restore && current.img.isConnected) current.img.src = current.restoreSrc;
    if (current.button?.isConnected) {
      current.button.disabled = false;
      setReviewCropButtonState(current.button, current.media);
    }
    URL.revokeObjectURL(current.originalUrl);
  }

  async function beginInlineCrop(stage, button) {
    if (active || button.disabled) return;
    const img = stage.querySelector('img');
    if (!img) return;

    button.disabled = true;
    const oldText = button.textContent;
    button.textContent = 'Opening crop…';
    try {
      const media = await resolveReviewMedia(stage);
      if (!media?.id) throw new Error('Could not identify this Rapid Review image.');
      if (media.media_type !== 'image') return;

      const originalBlob = await gmRequest({
        path:`/api/media/${encodeURIComponent(media.id)}/original`,
        responseType:'blob',
        timeout:300000,
      });
      const originalUrl = URL.createObjectURL(originalBlob);
      const restoreSrc = img.currentSrc || img.src;
      const crop = normalizeCrop(media);

      stage.dataset.naiReviewMediaId = media.id;
      img.src = originalUrl;
      try { await img.decode(); } catch (_) {}

      const overlay = document.createElement('div');
      overlay.className = 'naiReviewCropOverlayV253';
      overlay.innerHTML = `
        <div class="naiReviewCropMaskV253 top"></div>
        <div class="naiReviewCropMaskV253 bottom"></div>
        <div class="naiReviewCropHandleV253 top"></div>
        <div class="naiReviewCropHandleV253 bottom"></div>
      `;

      const controls = document.createElement('div');
      controls.className = 'naiReviewCropControlsV253';
      controls.innerHTML = `
        <div class="naiReviewCropReadoutV253"></div>
        <button type="button" data-reset>Reset</button>
        <button type="button" data-cancel>Cancel</button>
        <button type="button" class="primary" data-save>Save Crop</button>
      `;

      stage.classList.add('naiReviewCropActiveV253');
      stage.appendChild(overlay);
      stage.appendChild(controls);

      const topHandle = overlay.querySelector('.naiReviewCropHandleV253.top');
      const bottomHandle = overlay.querySelector('.naiReviewCropHandleV253.bottom');
      const topMask = overlay.querySelector('.naiReviewCropMaskV253.top');
      const bottomMask = overlay.querySelector('.naiReviewCropMaskV253.bottom');
      const readout = controls.querySelector('.naiReviewCropReadoutV253');

      const onMove = event => {
        if (!active?.drag) return;
        event.preventDefault();
        dragCrop(active.drag, event.clientY);
      };
      const onUp = () => { if (active) active.drag = null; };

      active = {
        stage, img, button, media, originalBlob, originalUrl, restoreSrc,
        overlay, controls, topHandle, bottomHandle, topMask, bottomMask, readout,
        top:crop.top, bottom:crop.bottom, drag:null, onMove, onUp, resizeObserver:null,
      };

      const startDrag = edge => event => {
        event.preventDefault();
        event.stopPropagation();
        if (!active) return;
        active.drag = edge;
        try { event.currentTarget.setPointerCapture?.(event.pointerId); } catch (_) {}
        dragCrop(edge, event.clientY);
      };
      topHandle.addEventListener('pointerdown', startDrag('top'));
      bottomHandle.addEventListener('pointerdown', startDrag('bottom'));
      window.addEventListener('pointermove', onMove, true);
      window.addEventListener('pointerup', onUp, true);

      if (typeof ResizeObserver === 'function') {
        active.resizeObserver = new ResizeObserver(placeOverlay);
        active.resizeObserver.observe(stage);
        active.resizeObserver.observe(img);
      }
      img.addEventListener('load', placeOverlay, {once:true});
      requestAnimationFrame(() => { placeOverlay(); paintActiveCrop(); });

      controls.querySelector('[data-reset]').addEventListener('click', event => {
        event.preventDefault(); event.stopPropagation();
        if (!active) return;
        active.top = 0; active.bottom = 0; paintActiveCrop();
      });
      controls.querySelector('[data-cancel]').addEventListener('click', event => {
        event.preventDefault(); event.stopPropagation();
        closeInlineCrop({restore:true});
      });
      controls.querySelector('[data-save]').addEventListener('click', async event => {
        event.preventDefault(); event.stopPropagation();
        if (!active) return;
        const current = active;
        const save = event.currentTarget;
        current.controls.classList.add('naiReviewCropBusyV253');
        save.disabled = true;
        try {
          const normalized = normalizeCrop({top:current.top, bottom:current.bottom});
          const updated = await gmRequest({
            method:'POST',
            path:`/api/media/${encodeURIComponent(current.media.id)}/crop`,
            headers:{'Content-Type':'application/json'},
            data:JSON.stringify(normalized),
          });
          Object.assign(current.media, updated || {});

          const previewBlob = await makeCroppedPreview(current.originalBlob, normalized);
          const previewUrl = URL.createObjectURL(previewBlob);
          const oldPreview = current.stage._naiV253CropPreviewUrl;
          if (oldPreview) URL.revokeObjectURL(oldPreview);
          current.stage._naiV253CropPreviewUrl = previewUrl;

          // Close without restoring, then show the saved crop in the Rapid Review image itself.
          const image = current.img;
          const stageRef = current.stage;
          const buttonRef = current.button;
          closeInlineCrop({restore:false});
          if (image.isConnected) image.src = previewUrl;
          if (buttonRef.isConnected) setReviewCropButtonState(buttonRef, updated || current.media);
          stageRef.dataset.naiReviewMediaId = current.media.id;
        } catch (err) {
          current.controls.classList.remove('naiReviewCropBusyV253');
          save.disabled = false;
          alert(err?.message || String(err));
        }
      });
    } catch (err) {
      alert(err?.message || String(err));
    } finally {
      if (button.isConnected) {
        button.disabled = !!active;
        if (!active && button.textContent === 'Opening crop…') button.textContent = oldText || 'Crop image';
        else if (active) button.textContent = 'Cropping…';
      }
    }
  }

  // Beat v2.5.2's old bubble listener at window capture. Cropping now stays entirely
  // inside Rapid Review and never selects/changes the normal viewer behind the modal.
  window.addEventListener('click', event => {
    const path = event.composedPath?.() || [];
    const button = path.find(node => node?.id === 'naiReviewCropBtn');
    if (!button || !path.includes(host)) return;
    event.preventDefault();
    event.stopImmediatePropagation();
    const stage = modalRoot.querySelector('#reviewStage');
    if (!stage || stage.querySelector('video')) return;
    beginInlineCrop(stage, button);
  }, true);

  // While inline crop mode is active, don't let Rapid Review shortcuts advance or
  // alter categories. Escape cancels the crop and returns to the same review item.
  window.addEventListener('keydown', event => {
    if (!active) return;
    if (event.key === 'Escape') {
      event.preventDefault();
      event.stopImmediatePropagation();
      closeInlineCrop({restore:true});
      return;
    }
    if ((event.key >= '1' && event.key <= '9') || event.key === 'Enter' || event.key === 'ArrowLeft' || event.key === 'ArrowRight') {
      event.preventDefault();
      event.stopImmediatePropagation();
    }
  }, true);

  // Prevent Rapid Review navigation/deletion/category changes/closing while the image is
  // actively being cropped. Only the inline crop controls remain interactive until Save or Cancel.
  window.addEventListener('click', event => {
    if (!active) return;
    const path = event.composedPath?.() || [];
    if (path.includes(active.overlay) || path.includes(active.controls)) return;
    const control = path.find(node => node?.matches?.(
      '#reviewNext, #reviewPrev, #reviewDelete, [data-close], .reviewSide input, .reviewSide button, .reviewCat'
    ));
    if (!control) return;
    event.preventDefault();
    event.stopImmediatePropagation();
  }, true);
})();
