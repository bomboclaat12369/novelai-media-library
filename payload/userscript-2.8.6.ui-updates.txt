// NovelAI Media Library v2.8.6: queue gallery and bulk image editing
(() => {
  'use strict';
  const host = document.getElementById('nai-media-host');
  const root = host?.shadowRoot;
  if (!root || document.documentElement.dataset.naiMediaV286Ui === '1') return;
  document.documentElement.dataset.naiMediaV286Ui = '1';

  const API = 'http://127.0.0.1:8765';
  const UI_STATE_KEY = 'nai-media-v1-ui';
  const modalRoot = root.getElementById('modalRoot');
  const catWrap = root.getElementById('categories');
  const charSelect = root.getElementById('characterSelect');
  if (!modalRoot || !catWrap || !charSelect) return;

  const style = document.createElement('style');
  style.id = 'nai-media-v286-ui-style';
  style.textContent = `
    .naiBulkEditModal285{width:min(1160px,96vw)!important;height:min(860px,94vh);}
    .naiBulkEditBody285{flex:1;min-height:0;overflow:auto;display:flex;flex-direction:column;gap:10px;padding:14px 16px!important}
    .naiBulkToolbar285{display:flex;align-items:center;gap:8px;flex-wrap:wrap}
    .naiBulkGallery285{min-height:120px;max-height:46vh;overflow:auto;display:grid;grid-template-columns:repeat(6,minmax(0,1fr));grid-auto-rows:max-content;gap:6px;padding:3px}
    .naiBulkCard285{position:relative;aspect-ratio:1/1;overflow:hidden;border-radius:7px;background:#0d1024;border:1px solid rgba(255,255,255,.1);cursor:pointer}
    .naiBulkCard285.selected{outline:2px solid #8278ff;outline-offset:1px}
    .naiBulkCard285 img{width:100%;height:100%;display:block;object-fit:cover}
    .naiBulkCard285 input{position:absolute;left:5px;top:5px;z-index:2;width:16px;height:16px}
    .naiBulkCard285 .name{position:absolute;left:0;right:0;bottom:0;padding:13px 4px 3px;background:linear-gradient(transparent,rgba(3,5,14,.94));font:600 8px/1.1 system-ui;white-space:nowrap;overflow:hidden;text-overflow:ellipsis}
    .naiBulkCard285 .review{position:absolute;right:4px;top:4px;padding:2px 4px;border-radius:4px;background:rgba(104,75,180,.9);font:700 8px/1 system-ui}
    .naiBulkEditor285{padding:10px;border:1px solid rgba(255,255,255,.1);border-radius:8px;background:#11142c}
    .naiBulkEmpty285{grid-column:1/-1;padding:30px 12px;text-align:center;color:#a7abc4;font:13px/1.5 system-ui}
    @media (max-width:900px){.naiBulkGallery285{grid-template-columns:repeat(4,minmax(0,1fr))}}
  `;
  root.appendChild(style);

  const escapeHtml = value => String(value ?? '').replace(/[&<>"']/g, c => ({'&':'&amp;','<':'&lt;','>':'&gt;','"':'&quot;',"'":'&#039;'}[c]));

  function request({method='GET', path, data=null, headers={}, responseType='text', timeout=120000}) {
    return new Promise((resolve, reject) => GM_xmlhttpRequest({
      method,
      url:`${API}${path}`,
      data,
      headers:{Origin:'https://novelai.net', ...headers},
      responseType,
      timeout,
      onload: response => {
        if (response.status >= 200 && response.status < 300) {
          if (responseType === 'blob') return resolve(response.response);
          const text = response.responseText || '';
          if (!text) return resolve(null);
          try { return resolve(JSON.parse(text)); } catch (_) { return resolve(text); }
        }
        let message = `Request failed (${response.status})`;
        try { message = JSON.parse(response.responseText || '{}').error || message; } catch (_) {}
        reject(new Error(message));
      },
      onerror: () => reject(new Error('Could not connect to the local NovelAI Media Library program.')),
      ontimeout: () => reject(new Error('The local media request timed out.')),
    }));
  }

  function readUiState() {
    try { return JSON.parse(localStorage.getItem(UI_STATE_KEY) || '{}') || {}; }
    catch (_) { return {}; }
  }

  function clearDeletedSelection(ids) {
    const deleted = new Set(ids);
    const saved = readUiState();
    if (deleted.has(saved.selectedMediaId)) saved.selectedMediaId = null;
    if (Array.isArray(saved.slotMediaIds)) saved.slotMediaIds = saved.slotMediaIds.map(id => deleted.has(id) ? null : id);
    localStorage.setItem(UI_STATE_KEY, JSON.stringify(saved));
  }

  function openBulkEditModal() {
    const characterId = charSelect.value || '';
    if (!characterId) return alert('Create or select a character first.');
    let library;
    let selected = new Set();
    let deleting = false;
    let wrap;
    let status;

    const close = () => { if (wrap?.isConnected) wrap.remove(); };
    const load = async () => {
      library = await request({path:'/api/library', timeout:15000});
      if (!Array.isArray(library?.media)) library.media = [];
      if (!Array.isArray(library?.characters)) library.characters = [];
      return library;
    };
    const character = () => (library?.characters || []).find(c => c.id === characterId) || null;
    const images = () => (library?.media || []).filter(m => m.character_id === characterId && m.media_type === 'image');
    const media = id => (library?.media || []).find(m => m.id === id) || null;
    const setStatus = text => { if (status) status.textContent = text; };

    const renderEditor = () => {
      const editor = wrap.querySelector('#naiBulkSingleEditor285');
      if (!editor) return;
      const chosen = [...selected].map(media).filter(Boolean);
      if (chosen.length !== 1) {
        editor.innerHTML = `<div class="muted">${chosen.length ? 'Select one image to edit its categories.' : 'Select an image to edit its categories, or select several to delete them together.'}</div>`;
        return;
      }
      const m = chosen[0];
      const c = character();
      editor.innerHTML = `
        <div style="font:700 13px system-ui;margin-bottom:6px">Edit selected image</div>
        <div class="fileSummary" style="margin-top:0">${escapeHtml(m.original_name || 'Selected image')}</div>
        <div class="muted" style="margin-top:10px">Categories</div>
        <div id="naiBulkCats285" class="checkGrid">${(c?.categories || []).map(cat => `<label class="check"><input type="checkbox" data-catid="${escapeHtml(cat.id)}" ${Array.isArray(m.categories) && m.categories.includes(cat.id) ? 'checked' : ''}> <span>${escapeHtml(cat.name)}</span></label>`).join('') || '<div class="muted">No custom categories.</div>'}</div>
        ${m.in_review ? '<div class="muted" style="margin-top:8px">This image is currently in Review and remains hidden from normal image categories.</div>' : ''}
        <button id="naiBulkSaveCats285" class="primary" style="margin-top:8px" ${c?.categories?.length ? '' : 'disabled'}>Save Categories</button>`;
      editor.querySelector('#naiBulkSaveCats285')?.addEventListener('click', async event => {
        const button = event.currentTarget;
        button.disabled = true;
        try {
          const categories = [...editor.querySelectorAll('#naiBulkCats285 input[data-catid]:checked')].map(input => input.dataset.catid);
          const updated = await request({method:'POST', path:`/api/media/${encodeURIComponent(m.id)}/categories`, headers:{'Content-Type':'application/json'}, data:JSON.stringify({categories})});
          Object.assign(m, updated || {categories});
          setStatus('Categories saved.');
        } catch (err) {
          setStatus(err?.message || String(err));
        } finally { if (button.isConnected) button.disabled = false; }
      });
    };

    const render = () => {
      const gallery = wrap.querySelector('#naiBulkGallery285');
      const allImages = images();
      for (const id of [...selected]) if (!allImages.some(m => m.id === id)) selected.delete(id);
      gallery.innerHTML = '';
      if (!allImages.length) {
        gallery.innerHTML = '<div class="naiBulkEmpty285">This character has no images.</div>';
      } else {
        const fragment = document.createDocumentFragment();
        for (const m of allImages) {
          const card = document.createElement('label');
          card.className = `naiBulkCard285${selected.has(m.id) ? ' selected' : ''}`;
          card.dataset.id = m.id;
          card.title = m.original_name || '';
          card.innerHTML = `<input type="checkbox" data-bulk-id="${escapeHtml(m.id)}" ${selected.has(m.id) ? 'checked' : ''}><img src="${API}/api/media/${encodeURIComponent(m.id)}/thumb" alt=""><div class="name">${escapeHtml(m.original_name || '')}</div>${m.in_review ? '<div class="review">Review</div>' : ''}`;
          fragment.appendChild(card);
        }
        gallery.appendChild(fragment);
      }
      const count = selected.size;
      wrap.querySelector('#naiBulkCount285').textContent = `${count} image${count === 1 ? '' : 's'} selected`;
      const deleteButton = wrap.querySelector('#naiBulkDelete285');
      deleteButton.disabled = !count || deleting;
      deleteButton.textContent = count ? `Delete selected images (${count})` : 'Delete selected images';
      renderEditor();
    };

    const deleteSelected = async () => {
      const ids = [...selected].filter(id => media(id));
      if (!ids.length || deleting) return;
      deleting = true;
      wrap.querySelectorAll('button').forEach(button => { button.disabled = true; });
      let removed = 0;
      const errors = [];
      for (const id of ids) {
        try {
          await request({method:'DELETE', path:`/api/media/${encodeURIComponent(id)}`});
          library.media = library.media.filter(item => item.id !== id);
          selected.delete(id);
          removed++;
          setStatus(`Deleted ${removed} of ${ids.length} selected image${ids.length === 1 ? '' : 's'}…`);
        } catch (err) { errors.push(err?.message || String(err)); }
      }
      clearDeletedSelection(ids.filter(id => !media(id)));
      deleting = false;
      render();
      setStatus(errors.length ? `Deleted ${removed}; ${errors.length} failed.` : `Deleted ${removed} image${removed === 1 ? '' : 's'}.`);
      root.getElementById('refreshBtn')?.click();
    };

    load().then(() => {
      const c = character();
      if (!c) throw new Error('Character not found.');
      wrap = document.createElement('div');
      wrap.className = 'modalWrap';
      wrap.innerHTML = `
        <div class="modal large naiBulkEditModal285">
          <div class="modalHead"><div class="modalTitle">Edit library — ${escapeHtml(c.name || '')}</div><button class="icon ghost" id="naiBulkClose285">×</button></div>
          <div class="modalBody naiBulkEditBody285">
            <div class="row"><div class="grow"><div style="font:700 13px system-ui">Select images</div><div class="muted" id="naiBulkCount285">0 images selected</div></div><button id="naiBulkAll285">Select all</button><button id="naiBulkNone285">Clear selection</button></div>
            <div class="muted">Choose one image to edit its categories, or select multiple images for one-click deletion.</div>
            <div class="naiBulkGallery285" id="naiBulkGallery285"></div>
            <div class="naiBulkEditor285" id="naiBulkSingleEditor285"></div>
            <div style="height:1px;background:rgba(255,255,255,.09);margin:4px 0"></div>
            <div style="font:700 13px system-ui">Delete a category</div>
            <div class="muted">Deleting a category removes its tag from images but does not delete the images.</div>
            ${(c.categories || []).length ? `<div class="row"><select id="naiBulkDeleteCategory285" class="grow">${c.categories.map(cat => `<option value="${escapeHtml(cat.id)}">${escapeHtml(cat.name)}</option>`).join('')}</select><button class="danger" id="naiBulkDeleteCategoryBtn285">Delete Category</button></div>` : '<div class="muted">This character has no custom categories.</div>'}
            <div class="status" id="naiBulkStatus285"></div>
          </div>
          <div class="modalFoot"><button id="naiBulkCloseFoot285">Close</button><button class="danger" id="naiBulkDelete285">Delete selected images</button></div>
        </div>`;
      modalRoot.innerHTML = '';
      modalRoot.appendChild(wrap);
      status = wrap.querySelector('#naiBulkStatus285');
      const closeAll = () => close();
      wrap.querySelector('#naiBulkClose285').addEventListener('click', closeAll);
      wrap.querySelector('#naiBulkCloseFoot285').addEventListener('click', closeAll);
      wrap.addEventListener('click', event => { if (event.target === wrap) closeAll(); });
      wrap.querySelector('#naiBulkAll285').addEventListener('click', () => { selected = new Set(images().map(m => m.id)); render(); });
      wrap.querySelector('#naiBulkNone285').addEventListener('click', () => { selected.clear(); render(); });
      wrap.querySelector('#naiBulkDelete285').addEventListener('click', deleteSelected);
      wrap.querySelector('#naiBulkGallery285').addEventListener('change', event => {
        const input = event.target.closest?.('input[data-bulk-id]');
        if (!input) return;
        if (input.checked) selected.add(input.dataset.bulkId); else selected.delete(input.dataset.bulkId);
        render();
      });
      const deleteCategory = wrap.querySelector('#naiBulkDeleteCategoryBtn285');
      deleteCategory?.addEventListener('click', async () => {
        const categoryId = wrap.querySelector('#naiBulkDeleteCategory285')?.value;
        if (!categoryId || deleting) return;
        const category = (character()?.categories || []).find(item => item.id === categoryId);
        if (!category || !confirm(`Delete the category “${category.name}”?\n\nIt will be removed from the images, but the images themselves will not be deleted.`)) return;
        deleteCategory.disabled = true;
        try {
          await request({method:'DELETE', path:`/api/categories/${encodeURIComponent(categoryId)}`});
          await load();
          close();
          root.getElementById('refreshBtn')?.click();
        } catch (err) {
          setStatus(err?.message || String(err));
          deleteCategory.disabled = false;
        }
      });
      render();
    }).catch(err => alert(err?.message || String(err)));
  }

  // Capture the existing Edit Images entry points before the original single-image
  // handler runs, while leaving the import and Review handlers untouched.
  root.addEventListener('click', event => {
    const target = event.target?.closest?.('#modalEditImages, #editImagesBtn');
    if (!target) return;
    event.preventDefault();
    event.stopImmediatePropagation();
    openBulkEditModal();
  }, true);

  // The queue-set dialog already contains every image in the active Rapid Review
  // queue. Add an explicit count so the full-gallery behavior is visible and obvious.
  function annotateQueueSetGallery() {
    const gallery = root.querySelector('#naiSetQueueGallery270');
    const title = root.querySelector('.naiSetQueueHead270 > div');
    if (!gallery || !title) return;
    const count = gallery.querySelectorAll('.naiSetQueueCard270').length;
    const next = `Assign all Rapid Review queue images to a set (${count})`;
    if (title.textContent !== next) title.textContent = next;
    const note = root.querySelector('.naiSetQueueSide270 .muted');
    if (note) note.textContent = `All ${count} images in this Rapid Review queue are shown on the right. Checked images will be added to the selected set.`;
  }
  let queueAnnotationQueued = false;
  function queueSetAnnotation() {
    if (queueAnnotationQueued) return;
    queueAnnotationQueued = true;
    queueMicrotask(() => { queueAnnotationQueued = false; annotateQueueSetGallery(); });
  }
  // openQueueSets appends its overlay directly to the shadow root, so observe only
  // direct children rather than the whole page or the whole application subtree.
  new MutationObserver(queueSetAnnotation).observe(root, {childList:true});
  queueSetAnnotation();
})();
